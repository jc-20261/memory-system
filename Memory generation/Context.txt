def get_category_chain(concrete_name: str) -> List[str]:
    chain = [concrete_name]
    current = concrete_name
    while current in CATEGORY_PARENTS and CATEGORY_PARENTS[current]:
        next_cat = CATEGORY_PARENTS[current][0]
        chain.append(next_cat)
        current = next_cat
    return chain

# ----------------------------------------------------------------------
# Object class (sparse storage)
# ----------------------------------------------------------------------
class Object:
    def __init__(self, name: str, template_name: str = None, obj_id: Optional[str] = None):
        self.name = name
        self.id = obj_id or f"{name}_{id(self)}"
        self.categories = get_category_chain(name)
        self.template_name = template_name or name
        self.overrides: Dict[str, Any] = {}
        self.spatial_relations: List[Tuple[str, str, str]] = []

    def add_attr(self, key: str, value: Any):
        self.overrides[key] = value

    def get_attr(self, key: str) -> Any:
        if key in self.overrides:
            return self.overrides[key]
        storage = ObjectStorage()
        defaults = storage.get_defaults(self.template_name)
        return defaults.get(key)

    @property
    def attributes(self) -> Dict[str, Any]:
        storage = ObjectStorage()
        merged = storage.get_defaults(self.template_name)
        merged.update(self.overrides)
        return merged

    def add_spatial(self, relation: str, other_obj: 'Object'):
        self.spatial_relations.append((self.name, relation, other_obj.name))

    def __repr__(self):
        return f"Object({self.name}, overrides={self.overrides})"

    _instance = None
    _defaults: Dict[str, Dict[str, Any]] = {}
    _required_attrs: Dict[str, Set[str]] = {}
    _parents: Dict[str, str] = {}

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def add_template(self, name: str, default_attributes: Dict[str, Any],
                     required_attrs: List[str] = None, parent: str = None):
        self._defaults[name] = default_attributes.copy()
        self._required_attrs[name] = set(required_attrs) if required_attrs else set()
        self._parents[name] = parent

    def _resolve_defaults(self, name: str) -> Dict[str, Any]:
        merged = {}
        if name in self._parents and self._parents[name]:
            parent_defaults = self._resolve_defaults(self._parents[name])
            merged.update(parent_defaults)
        if name in self._defaults:
            merged.update(self._defaults[name])
        return merged

    def _resolve_required_attrs(self, name: str) -> Set[str]:
        attrs = set()
        if name in self._parents and self._parents[name]:
            parent_req = self._resolve_required_attrs(self._parents[name])
            attrs.update(parent_req)
        if name in self._required_attrs:
            attrs.update(self._required_attrs[name])
        return attrs

    def get_defaults(self, name: str) -> Dict[str, Any]:
        return self._resolve_defaults(name)

    def get_required_attrs(self, name: str) -> Set[str]:
        return self._resolve_required_attrs(name)

    def instantiate(self, name: str, overrides: Dict[str, Any] = None) -> Object:
        if overrides:
            if name not in self._defaults:
                self._defaults[name] = {}
            for key in overrides.keys():
                if key not in self._defaults[name]:
                    self._defaults[name][key] = overrides[key]

        required = self._resolve_required_attrs(name)
        if overrides:
            missing = required - set(overrides.keys())
            if missing:
                raise ValueError(f"Missing required attributes for '{name}': {missing}")
        else:
            if required:
                raise ValueError(f"Required attributes for '{name}' not provided: {required}")

        instance = Object(name, template_name=name)
        if overrides:
            instance.overrides = overrides.copy()
        return instance

    def list_templates(self) -> List[str]:
        return list(self._defaults.keys())

@dataclass

# Action Storage (formerly ActionRepository)
# ----------------------------------------------------------------------
class ActionStorage:
    _instance = None
    _templates: Dict[str, ActionTemplate] = {}
    _name_embeddings: Dict[str, np.ndarray] = {}
    _embedding_model = None
    _flat_names_cache: Dict[str, List[str]] = {}

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            if SentenceTransformer:
                cls._embedding_model = SentenceTransformer("all-MiniLM-L6-v2")
        return cls._instance

    def add_template(self, template: ActionTemplate):
        self._templates[template.name] = template
        if template.name in self._flat_names_cache:
            del self._flat_names_cache[template.name]

    def get_template(self, name: str) -> Optional[ActionTemplate]:
        return self._templates.get(name)

    def instantiate(self, name: str, overrides: Dict[str, Any], auto_add: bool = True) -> Action:
        template = self._templates.get(name)
        if template is None:
            if auto_add:
                participants = overrides.get("participants", ["hand", "object"])
                cat = overrides.get("action_category", "INTERACTION" if len(participants) > 1 else "BODY_ONLY")
                new_template = ActionTemplate(
                    name=name,
                    default_duration=overrides.get("duration", 1),
                    action_category=cat,
                    default_participants=participants,
                    required_overrides=["changes"]
                )
                self.add_template(new_template)
                template = new_template
                print(f"✅ Auto‑added new action: {name} ({cat})")
            else:
                raise ValueError(f"Unknown action template: {name}")

        missing = set(template.required_overrides) - set(overrides.keys())
        if missing:
            raise ValueError(f"Missing required overrides for action '{name}': {missing}")

        provided_participants = overrides.get("participants", [])
        template_roles = template.default_participants
        if provided_participants:
            if len(provided_participants) != len(template_roles):
                raise ValueError(f"Participant mismatch: need {len(template_roles)}, got {len(provided_participants)}")
            binding = dict(zip(template_roles, provided_participants))
        else:
            binding = {}

        bound_kinematics = copy.deepcopy(template.kinematic_metadata)
        bound_kinematics["bound_target"] = binding.get("target_object", None)
        bound_kinematics["bound_hand"] = binding.get("hand", binding.get("arm", None))

        final_duration = overrides.get("duration", template.default_duration)
        final_tags = overrides.get("tags", template.default_tags)
        final_physical_changes = template.default_physical_changes
        final_physical_relations = template.default_physical_relations
        final_changes = overrides.get("changes", [])
        final_spatial_changes = overrides.get("spatial_changes", [])
        final_preconditions = template.preconditions

        sub_actions = []
        for sub_name in template.sub_action_templates:
            sub_defaults = template.sub_action_defaults.get(sub_name, {})
            sub_override = overrides.get("sub_actions", {}).get(sub_name, {})
            merged_sub = {**sub_defaults, **sub_override}
            sub_action = self.instantiate(sub_name, merged_sub, auto_add=False)
            sub_actions.append(sub_action)

        action = Action(
            name=name,
            duration=final_duration,
            participants=[],
            changes=final_changes,
            physical_changes=final_physical_changes,
            physical_relations=final_physical_relations,
            tags=final_tags,
            action_category=template.action_category,
            kinematic_metadata=bound_kinematics,
            kinematic_trajectory=template.kinematic_trajectory,
            preconditions=final_preconditions,
            sub_actions=sub_actions,
            temporal_type=template.temporal_type,
            _participant_names=provided_participants
        )
        return action

    def list_templates(self) -> List[str]:
        return list(self._templates.keys())

    def _get_embedding(self, text: str) -> np.ndarray:
        if self._embedding_model is None:
            return np.zeros(384)
        if text not in self._name_embeddings:
            self._name_embeddings[text] = self._embedding_model.encode(text, normalize_embeddings=True)
        return self._name_embeddings[text]

    def _similarity_between_names(self, name1: str, name2: str) -> float:
        if name1 == name2:
            return 1.0
        emb1 = self._get_embedding(name1)
        emb2 = self._get_embedding(name2)
        return float(np.dot(emb1, emb2))

    def _substitution_cost(self, name1: str, name2: str) -> float:
        if name1 == name2:
            return 0.0
        if self._embedding_model is None:
            return 1.0
        sim = self._similarity_between_names(name1, name2)
        if sim > 0.9:
            return 0.1
        elif sim > 0.8:
            return 0.3
        else:
            return 1.0

    def _weighted_edit_distance(self, seq1: List[str], seq2: List[str]) -> float:
        n, m = len(seq1), len(seq2)
        dp = [[0.0] * (m + 1) for _ in range(n + 1)]
        for i in range(n + 1):
            dp[i][0] = float(i)
        for j in range(m + 1):
            dp[0][j] = float(j)
        for i in range(1, n + 1):
            for j in range(1, m + 1):
                cost = self._substitution_cost(seq1[i-1], seq2[j-1])
                dp[i][j] = min(
                    dp[i-1][j] + 1.0,
                    dp[i][j-1] + 1.0,
                    dp[i-1][j-1] + cost
                )
        distance = dp[n][m]
        max_len = max(n, m)
        if max_len == 0:
            return 1.0
        return 1.0 - (distance / max_len)

    def _flatten_template_names(self, template: ActionTemplate) -> List[str]:
        if template.name in self._flat_names_cache:
            return self._flat_names_cache[template.name]
        if not template.sub_action_templates:
            result = [template.name]
        else:
            result = []
            for sub_name in template.sub_action_templates:
                sub_template = self._templates.get(sub_name)
                if sub_template:
                    result.extend(self._flatten_template_names(sub_template))
        self._flat_names_cache[template.name] = result
        return result

    def _flatten_action_names(self, action: Action) -> List[str]:
        if not action.sub_actions:
            return [action.name]
        result = []
        for sub in action.sub_actions:
            result.extend(self._flatten_action_names(sub))
        return result

    def get_parents(self, action_name: str) -> List[str]:
        if not hasattr(self, '_parent_map'):
            self._parent_map = self._build_parent_map()
        return self._parent_map.get(action_name, [])

    def _build_parent_map(self):
        parent_map = {}
        for name, template in self._templates.items():
            for sub in template.sub_action_templates:
                parent_map.setdefault(sub, []).append(name)
        return parent_map

    def find_similar_atomic_action(self, action: Action) -> Optional[Action]:
        if self._embedding_model is None:
            return None

        proposed = {
            "name": action.name,
            "participants": [p.name for p in action.participants],
            "trajectory": action.kinematic_trajectory,
            "category": action.action_category
        }

        best_match = None
        best_score = -1.0
        for name, template in self._templates.items():
            if template.sub_action_templates:
                continue
            score = self._compute_atomic_similarity(proposed, template)
            if score > best_score:
                best_score = score
                best_match = name

        if best_match and best_score > 0.0:
            return self.instantiate(best_match, {"changes": action.changes}, auto_add=False)
        return None

    def _compute_atomic_similarity(self, proposed: Dict, template: ActionTemplate) -> float:
        if self._similarity_between_names(proposed["name"], template.name) < 0.85:
            return 0.0
        if self._similarity_between_names(proposed.get("trajectory", "straight"), template.kinematic_trajectory) < 0.96:
            return 0.0
        if proposed.get("category", "INTERACTION") != template.action_category:
            return 0.0
        p1 = proposed.get("participants", [])
        p2 = template.default_participants
        if len(p1) != len(p2):
            return 0.0
        for a, b in zip(p1, p2):
            if self._similarity_between_names(a, b) < 0.96:
                return 0.0
        return 1.0

    def find_complex_template_by_flat_names(self, flat_names: List[str], threshold: float = 0.75) -> Optional[str]:
        if not flat_names:
            return None
        best_match = None
        best_score = -1.0

        for name, template in self._templates.items():
            if not template.sub_action_templates:
                continue
            template_flat_names = self._flatten_template_names(template)
            sim = self._weighted_edit_distance(flat_names, template_flat_names)
            if sim > best_score:
                best_score = sim
                best_match = name

        if best_match and best_score >= threshold:
            print(f"🔍 Found similar complex action: '{best_match}' (score: {best_score:.3f})")
            return best_match
        return None

    def compute_start_offsets(self, action: Action, parent_start: float = 0.0) -> float:
        action.start_offset = parent_start
        if not action.sub_actions:
            return action.duration

        if action.temporal_type == "sequential":
            current = parent_start
            for sub in action.sub_actions:
                child_dur = self.compute_start_offsets(sub, current)
                current += child_dur
            total_duration = current - parent_start
        elif action.temporal_type == "parallel":
            max_end = parent_start
            for sub in action.sub_actions:
                child_dur = self.compute_start_offsets(sub, parent_start)
                max_end = max(max_end, parent_start + child_dur)
            total_duration = max_end - parent_start
        else:
            total_duration = action.duration

        action.duration = int(round(total_duration))
        return total_duration

class Memory:
    def __init__(self, memory_id: str, protagonist: Protagonist):
        self.id = memory_id
        self.protagonist = protagonist
        self.objects: Dict[str, Object] = {protagonist.name: protagonist}
        self.actions: List[Action] = []
        self.timeline: List[int] = []
        # Extensions for lazy sub‑action addition
        self.extensions: List[Dict] = []
        self.extensions_applied: bool = False

    def add_object(self, obj: Object):
        self.objects[obj.name] = obj

    def add_action(self, action: Action):
        self.actions.append(action)
        self._apply_action(action)
        if self.timeline:
            self.timeline.append(self.timeline[-1] + action.duration)
        else:
            self.timeline.append(action.duration)

    def _apply_action(self, action: Action):
        for obj, attr, old, new in action.changes:
            if obj.get_attr(attr) == old or old is None:
                obj.overrides[attr] = new
        for obj1_name, rel, obj2_name, op in action.spatial_changes:
            if obj1_name in self.objects and obj2_name in self.objects:
                obj1 = self.objects[obj1_name]
                obj2 = self.objects[obj2_name]
                if op == "add":
                    if (obj1_name, rel, obj2_name) not in obj1.spatial_relations:
                        obj1.spatial_relations.append((obj1_name, rel, obj2_name))
                elif op == "remove":
                    obj1.spatial_relations = [r for r in obj1.spatial_relations
                                             if not (r[0]==obj1_name and r[1]==rel and r[2]==obj2_name)]
        for sub in action.sub_actions:
            self._apply_action(sub)

    def _find_action_by_name(self, name: str) -> Optional[Action]:
        for act in self.actions:
            found = self._search_action(act, name)
            if found:
                return found
        return None

    def _search_action(self, action: Action, name: str) -> Optional[Action]:
        if action.name == name:
            return action
        for sub in action.sub_actions:
            found = self._search_action(sub, name)
            if found:
                return found
        return None

    def apply_extensions(self, action_storage: 'ActionStorage', object_storage: 'ObjectStorage'):
        for ext in self.extensions:
            parent_name = ext["parent"]
            sub_action_data = ext["sub_action"]
            estimated_duration = ext.get("estimated_duration")

            parent_action = self._find_action_by_name(parent_name)
            if parent_action is None:
                continue

            if any(sub.name == sub_action_data["name"] for sub in parent_action.sub_actions):
                continue

            template = action_storage.get_template(sub_action_data["name"])
            if template is None:
                continue

            participants = []
            for p_name in sub_action_data.get("participants", []):
                obj = self.objects.get(p_name)
                if obj is None:
                    obj = Object(p_name)
                    self.objects[p_name] = obj
                participants.append(obj)

            overrides = {
                "participants": [p.name for p in participants],
                "duration": int(round(estimated_duration)) if estimated_duration else template.default_duration,
                "tags": sub_action_data.get("tags", [])
            }
            sub_action = action_storage.instantiate(sub_action_data["name"], overrides, auto_add=False)
            parent_action.sub_actions.append(sub_action)

        self.extensions_applied = True

    def ensure_extensions_applied(self, action_storage: 'ActionStorage', object_storage: 'ObjectStorage'):
        if not self.extensions_applied:
            self.apply_extensions(action_storage, object_storage)
            self.extensions_applied = True

    def export_initial_world_state(self) -> 'WorldState':
        ws = WorldState()
        for obj in self.objects.values():
            ws.add_object_state(obj.id, obj.attributes, obj.name)
            for rel in obj.spatial_relations:
                # rel is (obj.name, relation, other_obj.name)
                other = self.objects.get(rel[2])
                if other:
                    ws.add_spatial(obj.id, rel[1], other.id)
        return ws

    def get_state_at_time(self, time_sec: int) -> Dict[str, Object]:
        # Placeholder – would replay actions up to time_sec

# WorldState – Strict object identity, spatial graph, temporal, DSL
# ----------------------------------------------------------------------
class WorldState:
    # Define which spatial relations are inherently transitive
    TRANSITIVE_RELATIONS = {
        "above", "below", "left_of", "right_of",
        "front_of", "behind_of", "inside", "under", "over"
    }

    def __init__(self, objects: Optional[Dict[str, Dict[str, Any]]] = None,
                 spatial: Optional[List[Tuple[str, str, str]]] = None):
        self.objects = copy.deepcopy(objects) if objects else {}
        self.spatial_forward: Dict[str, Dict[str, Set[str]]] = defaultdict(lambda: defaultdict(set))
        self.spatial_reverse: Dict[str, Dict[str, Set[str]]] = defaultdict(lambda: defaultdict(set))
        if spatial:
            for subj, rel, obj in spatial:
                self.add_spatial(subj, rel, obj)
        self.current_time: float = 0.0
        self.name_to_ids: Dict[str, Set[str]] = defaultdict(set)

    # ---------- Object Management ----------
    def add_object_state(self, obj_id: str, attrs: Dict[str, Any], name: Optional[str] = None):
        if obj_id not in self.objects:
            self.objects[obj_id] = {}
        self.objects[obj_id].update(attrs)
        if name:
            self.name_to_ids[name].add(obj_id)

    def get_object_attrs(self, obj_id: str) -> Optional[Dict[str, Any]]:
        return self.objects.get(obj_id)

    def get_objects_by_name(self, name: str) -> Set[str]:
        return self.name_to_ids.get(name, set())

    def get_single_object_by_name(self, name: str) -> Optional[str]:
        ids = self.name_to_ids.get(name)
        if ids:
            return next(iter(ids))
        return None

    # ---------- Spatial Relations ----------
    def add_spatial(self, subj_id: str, rel: str, obj_id: str):
        self.spatial_forward[subj_id][rel].add(obj_id)
        self.spatial_reverse[obj_id][rel].add(subj_id)

    def remove_spatial(self, subj_id: str, rel: str, obj_id: str):
        if subj_id in self.spatial_forward and rel in self.spatial_forward[subj_id]:
            self.spatial_forward[subj_id][rel].discard(obj_id)
            if not self.spatial_forward[subj_id][rel]:
                del self.spatial_forward[subj_id][rel]
        if obj_id in self.spatial_reverse and rel in self.spatial_reverse[obj_id]:
            self.spatial_reverse[obj_id][rel].discard(subj_id)
            if not self.spatial_reverse[obj_id][rel]:
                del self.spatial_reverse[obj_id][rel]

    def get_direct_related(self, obj_id: str, rel: str, reverse: bool = False) -> Set[str]:
        if reverse:
            return self.spatial_reverse.get(obj_id, {}).get(rel, set())
        else:
            return self.spatial_forward.get(obj_id, {}).get(rel, set())

    # ---------- Transitive Inference (DFS) ----------
    def _dfs(self, start: str, rel: str, forward: bool) -> Set[str]:
        visited = set()
        result = set()
        stack = [start]
        while stack:
            current = stack.pop()
            if current in visited:
                continue
            visited.add(current)
            if forward:
                neighbours = self.spatial_forward.get(current, {}).get(rel, set())
            else:
                neighbours = self.spatial_reverse.get(current, {}).get(rel, set())
            for nbr in neighbours:
                if nbr not in visited:
                    result.add(nbr)
                    stack.append(nbr)
        return result

    def get_transitive_descendants(self, subj_id: str, rel: str) -> Set[str]:
        if rel not in self.TRANSITIVE_RELATIONS:
            return self.get_direct_related(subj_id, rel, reverse=False)
        return self._dfs(subj_id, rel, forward=True)

    def get_transitive_ancestors(self, obj_id: str, rel: str) -> Set[str]:
        if rel not in self.TRANSITIVE_RELATIONS:
            return self.get_direct_related(obj_id, rel, reverse=True)
        return self._dfs(obj_id, rel, forward=False)

    def is_transitively_related(self, subj_id: str, obj_id: str, rel: str) -> bool:
        return subj_id in self.get_transitive_ancestors(obj_id, rel)

    # ---------- Enhanced Preconditions (Mini-DSL) ----------
    def _evaluate_precondition(self, prec: Union[Dict, Tuple]) -> Tuple[bool, str]:
        if isinstance(prec, tuple):
            obj_name, attr, expected_val = prec
            op = "eq"
        else:
            obj_name = prec["object"]
            attr = prec["attribute"]
            expected_val = prec.get("value")
            op = prec.get("op", "eq")

        obj_ids = self.name_to_ids.get(obj_name)
        if not obj_ids:
            return False, f"Object '{obj_name}' not found in world state."
        obj_id = next(iter(obj_ids))
        obj_attrs = self.objects.get(obj_id)
        if obj_attrs is None:
            return False, f"Object '{obj_name}' (ID: {obj_id}) not found in world state."

        actual_val = obj_attrs.get(attr)
        if attr not in obj_attrs:
            return False, f"Attribute '{attr}' not found for object '{obj_name}'."

        if isinstance(expected_val, (int, float)) and isinstance(actual_val, (int, float)):
            pass
        elif isinstance(expected_val, str) and isinstance(actual_val, str):
            pass
        else:
            try:
                expected_val = float(expected_val)
                actual_val = float(actual_val)
            except (ValueError, TypeError):
                pass

        if op == "eq":
            if actual_val != expected_val:
                return False, f"{obj_name}.{attr} expected {expected_val}, got {actual_val}"
        elif op == "ne":
            if actual_val == expected_val:
                return False, f"{obj_name}.{attr} should not equal {expected_val}"
        elif op == "gt":
            if not (actual_val > expected_val):
                return False, f"{obj_name}.{attr} must be > {expected_val}, got {actual_val}"
        elif op == "gte":
            if not (actual_val >= expected_val):
                return False, f"{obj_name}.{attr} must be >= {expected_val}, got {actual_val}"
        elif op == "lt":
            if not (actual_val < expected_val):
                return False, f"{obj_name}.{attr} must be < {expected_val}, got {actual_val}"
        elif op == "lte":
            if not (actual_val <= expected_val):
                return False, f"{obj_name}.{attr} must be <= {expected_val}, got {actual_val}"
        elif op == "exists":
            if actual_val is None:
                return False, f"{obj_name}.{attr} does not exist"
        else:
            return False, f"Unknown operator '{op}' in precondition"

        return True, ""

    def check_preconditions(self, action: 'Action') -> Tuple[bool, str]:
        for prec in action.preconditions:
            ok, msg = self._evaluate_precondition(prec)
            if not ok:
                return False, f"Precondition failed for action '{action.name}': {msg}"
        return True, ""

    # ---------- Applying Effects ----------
    def apply_effects(self, action: 'Action'):
        for obj, attr, old, new in action.changes:
            obj_id = obj.id
            if obj_id not in self.objects:
                self.objects[obj_id] = {}
            self.objects[obj_id][attr] = new
            self.name_to_ids[obj.name].add(obj_id)

        for obj1, rel, obj2, op in action.spatial_changes:
            if op == "add":
                self.add_spatial(obj1.name, rel, obj2.name)
            elif op == "remove":
                self.remove_spatial(obj1.name, rel, obj2.name)

        for sub in action.sub_actions:
            self.apply_effects(sub)

    def apply_action(self, action: 'Action') -> Tuple[bool, str]:
        ok, msg = self.check_preconditions(action)
        if not ok:
            return False, msg

        self.apply_effects(action)
        self.current_time += action.duration
        return True, f"Applied '{action.name}' successfully, time now {self.current_time:.1f}s"

    # ---------- Bridge from Memory ----------
    def import_from_memory(self, memory: 'Memory'):
        for obj in memory.objects.values():
            self.add_object_state(obj.id, obj.attributes, obj.name)
            for rel in obj.spatial_relations:
                other = memory.objects.get(rel[2])
                if other:
                    self.add_spatial(obj.id, rel[1], other.id)

    # ---------- Copying ----------
    def copy(self) -> 'WorldState':
        new = WorldState()
        new.objects = copy.deepcopy(self.objects)
        new.spatial_forward = copy.deepcopy(self.spatial_forward)
        new.spatial_reverse = copy.deepcopy(self.spatial_reverse)
        new.current_time = self.current_time
        new.name_to_ids = copy.deepcopy(self.name_to_ids)
        return new

    def __repr__(self):
        return (f"WorldState(objects={len(self.objects)}, "
                f"spatial_forward={len(self.spatial_forward)}, "
                f"time={self.current_time:.1f}s)")


class RealWorldState(WorldState):
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance.objects = {}
            cls._instance.spatial_forward = defaultdict(lambda: defaultdict(set))
            cls._instance.spatial_reverse = defaultdict(lambda: defaultdict(set))
            cls._instance.current_time = 0.0
            cls._instance.name_to_ids = defaultdict(set)
        return cls._instance

    def reset(self):
        self.objects.clear()
        self.spatial_forward.clear()
        self.spatial_reverse.clear()
        self.current_time = 0.0
        self.name_to_ids.clear()


class HypotheticalWorldState(WorldState):

# BM25Index – Keyword search index
# ----------------------------------------------------------------------
class BM25Index:
    def __init__(self):
        self.memory_ids: List[Memory] = []
        self.tokenized_docs: List[List[str]] = []
        self.bm25 = None

    def _tokenize(self, text: str) -> List[str]:
        return [w.lower() for w in text.split() if len(w) > 2]

    def _memory_to_text(self, memory: Memory) -> str:
        parts = [f"Memory: {memory.id}"]
        for obj in memory.objects.values():
            attrs = " ".join([f"{k}={v}" for k, v in obj.attributes.items() if v is not None])
            if attrs:
                parts.append(f"Object {obj.name}: {attrs}")
        for act in memory.actions:
            parts.append(f"Action: {act.name} (tags: {' '.join(act.tags)})")
            for sub in act.sub_actions:
                parts.append(f"  Sub: {sub.name}")
        return " ".join(parts)

    def add_memory(self, memory: Memory):
        if BM25Okapi is None:
            return
        text = self._memory_to_text(memory)
        tokens = self._tokenize(text)
        self.memory_ids.append(memory)
        self.tokenized_docs.append(tokens)
        self.bm25 = BM25Okapi(self.tokenized_docs)

    def search(self, query: str, top_k: int = 20) -> List[Tuple[Memory, float]]:
        if not self.bm25 or not self.memory_ids:
            return []
        tokens = self._tokenize(query)
        scores = self.bm25.get_scores(tokens)
        top_indices = np.argsort(scores)[-top_k:][::-1]
        return [(self.memory_ids[i], scores[i]) for i in top_indices if scores[i] > 0]

# ----------------------------------------------------------------------
# VectorIndex – Semantic search index
# ----------------------------------------------------------------------
class VectorIndex:
    def __init__(self, model_name: str = "all-MiniLM-L6-v2"):
        self.model = SentenceTransformer(model_name) if SentenceTransformer else None
        self.memory_ids: List[Memory] = []
        self.embeddings: List[np.ndarray] = []

    def _memory_to_text(self, memory: Memory) -> str:
        parts = []
        for obj in memory.objects.values():
            attrs = " ".join([f"{k}={v}" for k, v in obj.attributes.items() if v is not None])
            if attrs:
                parts.append(f"{obj.name}: {attrs}")
        for act in memory.actions:
            parts.append(act.name)
            for sub in act.sub_actions:
                parts.append(sub.name)
        return " ".join(parts)

    def add_memory(self, memory: Memory):
        if self.model is None:
            return
        text = self._memory_to_text(memory)
        emb = self.model.encode(text, normalize_embeddings=True)
        self.memory_ids.append(memory)
        self.embeddings.append(emb)

    def search(self, query: str, top_k: int = 20) -> List[Tuple[Memory, float]]:
        if self.model is None or not self.memory_ids:
            return []
        query_emb = self.model.encode(query, normalize_embeddings=True)
        scores = np.dot(np.array(self.embeddings), query_emb)
        top_indices = np.argsort(scores)[-top_k:][::-1]
        return [(self.memory_ids[i], scores[i]) for i in top_indices if scores[i] > 0.1]

# Interactive Query Processor
# ----------------------------------------------------------------------
class InteractiveQueryProcessor:
    def __init__(self, search_engine: 'SearchEngine'):
        self.se = search_engine

    def extract_common_attributes(self, memories: List[Memory], top_k: int = 20) -> List[Tuple[str, Any, int]]:
        counter = Counter()
        for mem in memories:
            for obj in mem.objects.values():
                for attr, val in obj.attributes.items():
                    if val is not None:
                        counter[(attr, val)] += 1
        most_common = counter.most_common(top_k)
        return [(attr, val, count) for (attr, val), count in most_common]

    def generate_questions(self, common_attrs: List[Tuple[str, Any, int]], query_terms: List[str]) -> List[str]:
        questions = []
        query_set = set(query_terms)
        for attr, val, count in common_attrs:
            if any(term in attr or term in str(val) for term in query_set):
                continue
            questions.append(
                f"Many memories have '{attr} = {val}' (found in {count} memories). "
                f"Do you want to include this attribute in your search? (yes/no)"
            )
        return questions

    async def refine_search(self, original_query: str, feedback: Dict[str, bool]) -> List[Memory]:
        terms, goal = await self.se.query_processor.parse(original_query)
        for attr_val, include in feedback.items():
            if include:
                terms.append((attr_val, 2.0))
        results = self.se.search(terms)
        return [mem for mem, _ in results]

# ----------------------------------------------------------------------
# Query Processor (natural language parser)
# ----------------------------------------------------------------------
class QueryProcessor:
    def __init__(self, api_key: str = None, base_url: str = "https://api.deepseek.com", cache: bool = True):
        if api_key is None:
            api_key = os.getenv("DEEPSEEK_API_KEY")
        if AsyncOpenAI is None:
            raise ImportError("openai library not installed. Please install: pip install openai")
        self.client = AsyncOpenAI(api_key=api_key, base_url=base_url)
        self.cache = {} if cache else None
        self.system_prompt = """
You are a query translation engine. Given a natural language question about a memory or a planning request,
extract:
1. A list of search terms (with weights) to find relevant memories.
2. A goal state (optional) that the user wants to achieve. The goal state is a dict of attribute-value pairs
   for some object, e.g., {"peach": {"cut_state": "halved", "pit_present": false}}.

Output JSON with:
{
  "search_terms": [{"term": "twist", "weight": 2.0}, ...],
  "goal_state": {"object_name": {"attribute": "value", ...}}   // or null if no goal
}
If the query is a planning question (e.g., "How do I cut a peach?"), include a goal_state.
If it's a general memory query (e.g., "Show me memories of peeling"), just provide search_terms.
"""

    async def parse(self, query: str) -> Tuple[List[Tuple[str, float]], Optional[Dict[str, Dict[str, Any]]]]:
        if self.cache and query in self.cache:
            return self.cache[query]
        try:
            response = await self.client.chat.completions.create(
                model="deepseek-v4-flash",
                messages=[
                    {"role": "system", "content": self.system_prompt},
                    {"role": "user", "content": query}
                ],
                response_format={"type": "json_object"},
                temperature=0.3,
                max_tokens=500
            )
            data = json.loads(response.choices[0].message.content)
            terms = [(item["term"], item.get("weight", 1.0)) for item in data.get("search_terms", [])]
            goal = data.get("goal_state")
            result = (terms, goal)
        except Exception as e:
            terms = self._fallback_parse(query)
            goal = None
            result = (terms, goal)
        if self.cache:
            self.cache[query] = result
        return result

    def _fallback_parse(self, query: str) -> List[Tuple[str, float]]:
        words = query.lower().split()
        stopwords = {"what", "does", "the", "of", "a", "an", "to", "for", "with", "on", "at", "from", "by", "in", "is", "are"}
        terms = [(w, 1.0) for w in words if w not in stopwords and len(w) > 2]
        if "rotate" in query or "twist" in query:
            terms.append(("rotate", 2.0))
        if "grasp" in query:
            terms.append(("grasp", 2.0))
        if "peel" in query:
            terms.append(("peel", 2.0))
        if "cut" in query:
            terms.append(("cut", 2.0))
        if "pit" in query:
            terms.append(("pit", 2.0))
        return terms

# ----------------------------------------------------------------------
# Search Engine (Memory Storage)
# ----------------------------------------------------------------------
class SearchEngine:
    def __init__(self, max_memories: int = 10, use_hybrid: bool = False,
                 action_storage: ActionStorage = None, object_storage: ObjectStorage = None,
                 query_processor: QueryProcessor = None):
        self.max_memories = max_memories
        self.memories: deque = deque(maxlen=max_memories)
        self.index: Dict[str, List[Tuple[Memory, float]]] = {}
        self.use_hybrid = use_hybrid
        self.action_storage = action_storage
        self.object_storage = object_storage
        self.query_processor = query_processor
        if self.use_hybrid:
            self.bm25_index = BM25Index()
            self.vector_index = VectorIndex()

    def add_memory(self, mem: Memory):
        if self.action_storage and self.object_storage:
            mem.ensure_extensions_applied(self.action_storage, self.object_storage)
        if len(self.memories) >= self.max_memories:
            oldest = self.memories[0]
            self._remove_from_index(oldest)
            if self.use_hybrid:
                self._rebuild_hybrid_indices()
        self.memories.append(mem)
        self._index_memory(mem)
        if self.use_hybrid:
            self.bm25_index.add_memory(mem)
            self.vector_index.add_memory(mem)

    def _remove_from_index(self, mem: Memory):
        for term, entries in list(self.index.items()):
            self.index[term] = [(m, w) for m, w in entries if m != mem]
            if not self.index[term]:
                del self.index[term]

    def _rebuild_hybrid_indices(self):
        self.bm25_index = BM25Index()
        self.vector_index = VectorIndex()
        for mem in self.memories:
            self.bm25_index.add_memory(mem)
            self.vector_index.add_memory(mem)

    def _index_memory(self, mem: Memory):
        term_weights: Dict[str, float] = {}

        def index_action(action: Action, weight_scale: float = 1.0):
            term_weights[action.name] = term_weights.get(action.name, 0.0) + 2.0 * weight_scale
            for tag in action.tags:
                term_weights[tag] = term_weights.get(tag, 0.0) + 2.0 * weight_scale
            for obj, attr, old, new in action.physical_changes:
                term = f"{obj.name}.{attr}={new}"
                term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
            for actor, rel, target in action.physical_relations:
                term = f"{actor} {rel} {target}"
                term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
            meta = action.kinematic_metadata
            for key, val in meta.items():
                if isinstance(val, str):
                    term = f"{key}={val}"
                    term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
                elif isinstance(val, dict):
                    for k2, v2 in val.items():
                        term = f"{key}_{k2}={v2}"
                        term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
            if "bound_target" in meta and meta["bound_target"]:
                term = f"target={meta['bound_target']}"
                term_weights[term] = term_weights.get(term, 0.0) + 2.0 * weight_scale
            for sub in action.sub_actions:
                index_action(sub, weight_scale * 0.8)

        for obj in mem.objects.values():
            for attr, val in obj.attributes.items():
                if val is not None:
                    term = f"{attr}={val}"
                    term_weights[term] = term_weights.get(term, 0.0) + 1.0
                    for i, cat in enumerate(obj.categories):
                        cat_term = f"{attr}={val} ({cat})"
                        weight = 1.0 / (i + 2)
                        term_weights[cat_term] = term_weights.get(cat_term, 0.0) + weight

        for obj in mem.objects.values():
            for rel in obj.spatial_relations:
                spatial_term = f"{rel[0]} {rel[1]} {rel[2]}"
                term_weights[spatial_term] = term_weights.get(spatial_term, 0.0) + 1.0

        for act in mem.actions:
            index_action(act)

        for term, weight in term_weights.items():
            self.index.setdefault(term, []).append((mem, weight))

    def search(self, query_terms: List[Tuple[str, float]]) -> List[Tuple[Memory, float]]:
        scores = {}
        for term, q_weight in query_terms:
            for mem, idx_weight in self.index.get(term, []):
                scores[mem] = scores.get(mem, 0.0) + idx_weight * q_weight
        return sorted(scores.items(), key=lambda x: x[1], reverse=True)

    def search_hybrid(self, query_text: str, top_k: int = 20) -> List[Tuple[Memory, float]]:
        if not self.use_hybrid or np is None or BM25Okapi is None:
            simple_terms = [(w, 1.0) for w in query_text.split() if len(w) > 2]
            return self.search(simple_terms)

        terms = [(w, 1.0) for w in query_text.split() if len(w) > 2]
        keyword_results = self.search(terms)
        bm25_results = self.bm25_index.search(query_text, top_k=top_k)
        vector_results = self.vector_index.search(query_text, top_k=top_k)

        all_memories = set()
        for mem, _ in keyword_results:
            all_memories.add(mem)
        for mem, _ in bm25_results:
            all_memories.add(mem)
        for mem, _ in vector_results:
            all_memories.add(mem)

        rank_maps = {
            'keyword': {mem: idx for idx, (mem, _) in enumerate(keyword_results)},
            'bm25': {mem: idx for idx, (mem, _) in enumerate(bm25_results)},
            'vector': {mem: idx for idx, (mem, _) in enumerate(vector_results)}
        }

        fused_scores = {}
        for mem in all_memories:
            score = 0.0
            for system in ['keyword', 'bm25', 'vector']:
                rank = rank_maps[system].get(mem, 1000)
                score += 1.0 / (rank + 60)
            fused_scores[mem] = score

        return sorted(fused_scores.items(), key=lambda x: x[1], reverse=True)

    def _collect_atomic_actions(self, action: Action) -> List[Action]:
        if not action.sub_actions:
            return [action]
        result = []
        for sub in action.sub_actions:
            result.extend(self._collect_atomic_actions(sub))
        return result

    def _flatten_memory_actions(self, memory: Memory) -> List[Action]:
        result = []
        for act in memory.actions:
            result.extend(self._collect_atomic_actions(act))
        return result

    def _get_action_duration_in_memory(self, memory: Memory, action_name: str) -> Optional[float]:
        if not hasattr(memory, '_flattened_actions'):
            memory._flattened_actions = self._flatten_memory_actions(memory)
        for act in memory._flattened_actions:
            if act.name == action_name:
                return act.duration
        return None

    async def search_with_temporal_proxy(
        self,
        query_text: str,
        processor: QueryProcessor,
        top_k: int = 100,
        temporal_weight: float = 0.3,
        refine_with_llm: bool = False,
        refine_count: int = 10,
        decomposer: 'ActionDecomposer' = None,
        object_storage: ObjectStorage = None
    ) -> List[Tuple[Memory, float]]:
        # 1. Parse query
        terms, goal = await processor.parse(query_text)

        # 2. Identify target action
        target_action = None
        if self.action_storage:
            for action_name in self.action_storage.list_templates():
                if action_name in query_text.lower():
                    target_action = action_name
                    break

        # 3. Structural search
        if self.use_hybrid:
            candidate_results = self.search_hybrid(query_text, top_k=top_k)
        else:
            simple_terms = [(w, 1.0) for w in query_text.split() if len(w) > 2]
            candidate_results = self.search(simple_terms)

        if not target_action:
            return candidate_results

        # 4. Ensure extensions are applied and compute proxy scores
        for mem, struct_score in candidate_results:
            if self.action_storage and self.object_storage:
                mem.ensure_extensions_applied(self.action_storage, self.object_storage)
            ancestor = target_action
            dur = self._get_action_duration_in_memory(mem, ancestor)
            while dur is None:
                parents = self.action_storage.get_parents(ancestor) if self.action_storage else []
                if not parents:
                    break
                ancestor = parents[0]
                dur = self._get_action_duration_in_memory(mem, ancestor)
            if dur is not None and dur > 0:
                proxy_score = 1.0 / (dur + 0.01)
            else:
                proxy_score = 0.0
            mem._combined_score = struct_score * (1 - temporal_weight) + proxy_score * temporal_weight

        candidate_results.sort(key=lambda x: x[0]._combined_score, reverse=True)

        # 5. LLM refinement (optional)
        if refine_with_llm and decomposer and object_storage:
            top_candidates = candidate_results[:refine_count]
            for mem, _ in top_candidates:
                if not self._action_exists_in_memory(mem, target_action):
                    await decomposer.instantiate_action_with_llm(
                        mem, target_action, object_storage, self.action_storage
                    )
            for mem, _ in top_candidates:
                llm_action = getattr(mem, '_llm_instantiated_actions', {}).get(target_action)
                mem._llm_duration = llm_action.duration if llm_action else None
            top_candidates.sort(key=lambda x: x[0]._llm_duration if x[0]._llm_duration else float('inf'))
            return top_candidates

        return candidate_results

    def _action_exists_in_memory(self, memory: Memory, action_name: str) -> bool:
        flattened = self._flatten_memory_actions(memory)
        return any(a.name == action_name for a in flattened)

    async def search_natural(
        self,
        query_text: str,
        processor: QueryProcessor,
        use_temporal_proxy: bool = True,
        refine_with_llm: bool = False,
        use_interactive: bool = False,
        decomposer: 'ActionDecomposer' = None,
        object_storage: ObjectStorage = None
    ) -> Tuple[List[Tuple[Memory, float]], Optional[Dict]]:
        if use_temporal_proxy and self.action_storage is not None:
            results = await self.search_with_temporal_proxy(
                query_text, processor,
                top_k=100,
                temporal_weight=0.3,
                refine_with_llm=refine_with_llm,
                refine_count=10,
                decomposer=decomposer,
                object_storage=object_storage
            )
            _, goal = await processor.parse(query_text)

            if use_interactive and results:
                interactive = InteractiveQueryProcessor(self)
                top_memories = [mem for mem, _ in results[:1000]]
                common = interactive.extract_common_attributes(top_memories, top_k=20)
                questions = interactive.generate_questions(common, query_text.split())
                print("\n📋 Interactive Query Refinement:")
                for q in questions[:5]:
                    print(f"  - {q}")
                # In a real system, capture user feedback here.

            return results, goal
        else:
            terms, goal = await processor.parse(query_text)
            if self.use_hybrid:
                results = self.search_hybrid(query_text)
            else:
                results = self.search(terms)
            return results, goal

    def get_all_memories(self) -> List[Memory]:
        return list(self.memories)

# ----------------------------------------------------------------------
# Planner (Backward, Hierarchical, Plan Validation)
# ----------------------------------------------------------------------
class Planner:
    def __init__(self, action_storage: ActionStorage, object_storage: ObjectStorage, search_engine: SearchEngine):
        self.action_storage = action_storage
        self.object_storage = object_storage
        self.se = search_engine

    def _get_effects(self, template: ActionTemplate) -> Set[Tuple[str, str, Any]]:
        effects = set()
        for sub_name, sub_defaults in template.sub_action_defaults.items():
            for change in sub_defaults.get("changes", []):
                obj = change.get("object")
                attr = change.get("attribute")
                new = change.get("new")
                if obj and attr and new is not None:
                    effects.add((obj, attr, new))
        return effects

    def plan_backward(self, goal_state: Dict[str, Dict[str, Any]], max_depth: int = 10) -> List[Action]:
        plan = []
        achieved = set()

        def satisfy_goal(goal_dict: Dict[str, Dict[str, Any]], depth: int) -> bool:
            if depth == 0:
                return False
            for obj_name, attr_dict in goal_dict.items():
                for attr, val in attr_dict.items():
                    candidates = []
                    for act_name, template in self.action_storage._templates.items():
                        effects = self._get_effects(template)
                        if (obj_name, attr, val) in effects:
                            candidates.append(act_name)
                    for act_name in candidates:
                        template = self.action_storage._templates[act_name]
                        participants = template.default_participants
                        if not participants:
                            continue
                        target_role = "target_object" if "target_object" in participants else participants[-1]
                        participant_map = {role: obj_name if role == target_role else role for role in participants}
                        overrides = {
                            "participants": [participant_map.get(p, p) for p in participants],
                            "changes": [{"object": obj_name, "attribute": attr, "old": None, "new": val}]
                        }
                        try:
                            action = self.action_storage.instantiate(act_name, overrides, auto_add=False)
                            pre_goal = {}
                            for prec in action.preconditions:
                                if isinstance(prec, dict):
                                    obj_name2 = prec.get("object")
                                    attr2 = prec.get("attribute")
                                    val2 = prec.get("value")
                                    if obj_name2 and attr2 and val2 is not None:
                                        pre_goal.setdefault(obj_name2, {})[attr2] = val2
                            if pre_goal:
                                sub_plan = self.plan_backward(pre_goal, depth-1)
                                if sub_plan is None:
                                    continue
                                plan.extend(sub_plan)
                            plan.append(action)
                            achieved.add((obj_name, attr, val))
                            return True
                        except Exception as e:
                            continue
            return False

        for depth in range(1, max_depth+1):
            if satisfy_goal(goal_state, depth):
                plan.reverse()
                return plan
        return None

    def plan_hierarchical(self, goal_state: Dict[str, Dict[str, Any]], max_depth: int = 10) -> List[Action]:
        for act_name, template in self.action_storage._templates.items():
            effects = self._get_effects(template)
            goal_set = set()
            for obj, attrs in goal_state.items():
                for attr, val in attrs.items():
                    goal_set.add((obj, attr, val))
            if goal_set.issubset(effects):
                participants = template.default_participants
                target_role = "target_object" if "target_object" in participants else (participants[-1] if participants else None)
                if target_role is None:
                    continue
                obj_name = next(iter(goal_state.keys()))
                participant_map = {role: obj_name if role == target_role else role for role in participants}
                overrides = {
                    "participants": [participant_map.get(p, p) for p in participants],
                    "changes": [{"object": obj_name, "attribute": attr, "old": None, "new": val} for attr, val in goal_state[obj_name].items()]
                }
                try:
                    complex_action = self.action_storage.instantiate(act_name, overrides, auto_add=False)
                    plan = []
                    for sub in complex_action.sub_actions:
                        sub_goal = {}
                        for obj, attr, old, new in sub.changes:
                            if new is not None:
                                sub_goal.setdefault(obj.name, {})[attr] = new
                        if sub_goal:
                            sub_plan = self.plan_backward(sub_goal, max_depth-1)
                            if sub_plan is None:
                                continue
                            plan.extend(sub_plan)
                        else:
                            plan.append(sub)
                    return plan
                except Exception as e:
                    continue
        return self.plan_backward(goal_state, max_depth)

    def validate_plan(self, plan: List[Action], base_state: WorldState,
                      time_limit: Optional[float] = None) -> Tuple[bool, List[str]]:
        state = base_state.copy()
        errors = []
        for i, action in enumerate(plan):
            ok, msg = state.apply_action(action)
            if not ok:
                errors.append(f"Step {i+1}: {msg}")
                return False, errors
        if time_limit is not None and state.current_time > time_limit:
            errors.append(f"Plan exceeded time limit: {state.current_time:.1f}s > {time_limit:.1f}s")
            return False, errors
        return True, []

# Action Decomposer (LLM-driven)
# ----------------------------------------------------------------------
class ActionDecomposer:
    def __init__(self, api_key: str = None):
        if api_key is None:
            api_key = os.getenv("DEEPSEEK_API_KEY")
        self.client = AsyncOpenAI(api_key=api_key, base_url="https://api.deepseek.com")
        self.default_instructions = """
You are a physical, mechanical, and biomechanical action decomposition engine. 
Given a high‑level action, break it into the smallest, most atomic physical sub‑actions.

Guidelines:
- If the action is already atomic (e.g., "rotate", "pull", "push", "grind", "fall"), return an empty list.
- Each sub‑action must be a single, indivisible movement, force application, or state transition.
- Use only the provided participant names (or parts of them) in the sub‑actions.
- Each sub‑action should have a reasonable duration (1–5 seconds) from the list: [1, 3, 10, 20, 60, ...].
- Output a JSON object with a single key "sub_actions", which is a list of objects:
  [{"name": "...", "duration": 1, "participants": [...], "kinematic_trajectory": "...", "temporal_type": "sequential"}]
- The overall temporal_type for this action is "sequential".

If no decomposition, return {"sub_actions": []}.
"""

    async def decompose_action_tree(self, action: Action, action_storage: ActionStorage, object_storage: ObjectStorage) -> Action:
        # 1. Recursively decompose children first
        if action.sub_actions:
            decomposed_children = []
            for sub in action.sub_actions:
                decomposed_sub = await self.decompose_action_tree(sub, action_storage, object_storage)
                decomposed_children.append(decomposed_sub)
            action.sub_actions = decomposed_children

        # 2. If no children (atomic), try atomic matching or LLM decomposition
        if not action.sub_actions:
            matched_atomic = action_storage.find_similar_atomic_action(action)
            if matched_atomic:
                print(f"♻️ Reusing atomic action '{matched_atomic.name}' for '{action.name}'")
                action.name = matched_atomic.name
                action.duration = matched_atomic.duration
                action.participants = matched_atomic.participants
                action.physical_changes = matched_atomic.physical_changes
                action.physical_relations = matched_atomic.physical_relations
                action.kinematic_metadata = matched_atomic.kinematic_metadata
                action.temporal_type = matched_atomic.temporal_type
                action.tags = list(set(action.tags + matched_atomic.tags))
                return action
            else:
                print(f"🔍 Asking LLM to decompose: {action.name}")
                sub_data = await self._call_llm_for_decomposition(action)
                if sub_data:
                    instantiated_subs = []
                    for sub_d in sub_data:
                        resolved_participants = self._resolve_participants(
                            sub_d.get("participants", []),
                            action.participants,
                            object_storage
                        )
                        sub_action = Action(
                            name=sub_d["name"],
                            duration=sub_d.get("duration", 1),
                            participants=resolved_participants,
                            changes=[],
                            physical_changes=[],
                            physical_relations=[],
                            tags=[sub_d["name"]],
                            action_category="INTERACTION",
                            kinematic_metadata={},
                            kinematic_trajectory=sub_d.get("kinematic_trajectory", "straight"),
                            temporal_type=sub_d.get("temporal_type", "sequential")
                        )
                        decomposed_sub = await self.decompose_action_tree(sub_action, action_storage, object_storage)
                        instantiated_subs.append(decomposed_sub)
                    action.sub_actions = instantiated_subs
                else:
                    print(f"✅ Adding new atomic action: {action.name}")
                    self._register_atomic_action(action, action_storage)
                    return action

        # 3. Complex action matching (after children are resolved)
        if action.sub_actions:
            flat_names = action_storage._flatten_action_names(action)
            matched_name = action_storage.find_complex_template_by_flat_names(flat_names, threshold=0.75)
            if matched_name:
                print(f"♻️ Reusing complex action '{matched_name}' for '{action.name}'")
                template = action_storage.get_template(matched_name)
                new_action = action_storage.instantiate(matched_name, {
                    "participants": action._participant_names,
                    "changes": action.changes,
                    "duration": action.duration,
                    "tags": action.tags
                }, auto_add=False)
                action.duration = new_action.duration
                action.sub_actions = new_action.sub_actions
                action.physical_changes = new_action.physical_changes
                action.physical_relations = new_action.physical_relations
                action.kinematic_metadata = new_action.kinematic_metadata
                action.temporal_type = new_action.temporal_type
                return action
            else:
                print(f"✅ Adding new complex action: {action.name}")
                self._register_complex_action(action, action_storage)
                return action

        return action

    async def _call_llm_for_decomposition(self, action: Action) -> List[Dict]:
        participant_names = [p.name for p in action.participants]
        meta = action.kinematic_metadata
        prompt = f"""
Action name: {action.name}
Participants: {participant_names}
Kinematic metadata: {meta}
Current physical changes: {action.changes}
Tags: {action.tags}

{self.default_instructions}
"""
        try:
            response = await self.client.chat.completions.create(
                model="deepseek-v4-flash",
                messages=[
                    {"role": "system", "content": "You are an expert in physical, mechanical, and biomechanical task decomposition."},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"},
                temperature=0.3,
                max_tokens=500
            )
            data = json.loads(response.choices[0].message.content)
            return data.get("sub_actions", [])
        except Exception as e:
            print(f"LLM decomposition failed for {action.name}: {e}")
            return []

    def _resolve_participants(self, llm_names: List[str], parent_participants: List[Object], object_storage: ObjectStorage) -> List[Object]:
        resolved = []
        for llm_name in llm_names:
            matched = None
            for p in parent_participants:
                if p.name == llm_name:
                    matched = p
                    break
            if matched:
                resolved.append(matched)
                continue
            try:
                new_obj = object_storage.instantiate(llm_name, {})
                resolved.append(new_obj)
            except Exception:
                new_obj = Object(llm_name)
                resolved.append(new_obj)
        return resolved

    def _register_atomic_action(self, action: Action, action_storage: ActionStorage):
        template = ActionTemplate(
            name=action.name,
            default_duration=action.duration,
            action_category=action.action_category,
            default_participants=[p.name for p in action.participants],
            default_tags=action.tags,
            kinematic_metadata=action.kinematic_metadata,
            kinematic_trajectory=action.kinematic_trajectory,
            preconditions=action.preconditions,
            required_overrides=["changes"],
            sub_action_templates=[],
            sub_action_defaults={},
            temporal_type="atomic"
        )
        action_storage.add_template(template)

    def _register_complex_action(self, action: Action, action_storage: ActionStorage):
        sub_names = []
        sub_defaults = {}
        for sub in action.sub_actions:
            sub_names.append(sub.name)
            sub_defaults[sub.name] = {
                "participants": [p.name for p in sub.participants],
                "changes": sub.changes,
                "duration": sub.duration,
                "tags": sub.tags,
                "kinematic_trajectory": sub.kinematic_trajectory
            }
        template = ActionTemplate(
            name=action.name,
            default_duration=action.duration,
            action_category=action.action_category,
            default_participants=[p.name for p in action.participants],
            default_tags=action.tags,
            kinematic_metadata=action.kinematic_metadata,
            kinematic_trajectory=action.kinematic_trajectory,
            preconditions=action.preconditions,
            required_overrides=["changes"],
            sub_action_templates=sub_names,
            sub_action_defaults=sub_defaults,
            temporal_type="sequential"
        )
        action_storage.add_template(template)

    def _flatten_memory(self, memory: Memory) -> List[Action]:
        result = []
        for act in memory.actions:
            result.extend(self._flatten_action(act))
        return result

    def _flatten_action(self, action: Action) -> List[Action]:
        if not action.sub_actions:
            return [action]
        result = []
        for sub in action.sub_actions:
            result.extend(self._flatten_action(sub))
        return result

    async def instantiate_action_with_llm(
        self,
        memory: Memory,
        action_name: str,
        object_storage: ObjectStorage,
        action_storage: ActionStorage
    ) -> Optional[Action]:
        memory.ensure_extensions_applied(action_storage, object_storage)

        flattened = self._flatten_memory(memory)
        existing = next((a for a in flattened if a.name == action_name), None)
        if existing:
            return existing

        ancestor = action_name
        ancestor_duration = None
        ancestor_participants = []
        while True:
            parents = action_storage.get_parents(ancestor)
            if not parents:
                break
            parent = parents[0]
            parent_action = memory._find_action_by_name(parent)
            if parent_action:
                ancestor = parent
                ancestor_duration = parent_action.duration
                ancestor_participants = [p.name for p in parent_action.participants]
                break
            ancestor = parent
        if ancestor_duration is None:
            template = action_storage.get_template(action_name)
            if template:
                participants = [p.name for p in memory.objects.values() if 'hand' in p.name or 'arm' in p.name]
                overrides = {"participants": participants[:len(template.default_participants)]}
                return action_storage.instantiate(action_name, overrides, auto_add=False)
            return None

        memory_text = f"Memory ID: {memory.id}\n"
        for act in memory.actions:
            memory_text += f"Action: {act.name} (duration {act.duration}s, tags: {act.tags})\n"
            for sub in act.sub_actions:
                memory_text += f"  Sub: {sub.name} (duration {sub.duration}s)\n"

        prompt = f"""
You are an expert in estimating the duration of an action based on context.
We have a memory that contains the action '{ancestor}' which took {ancestor_duration} seconds.
We want to estimate the duration of the sub‑action '{action_name}', which is part of '{ancestor}'.

Memory context:
{memory_text}

Participants of '{ancestor}': {ancestor_participants}
Template for '{action_name}': {action_storage.get_template(action_name).kinematic_metadata if action_storage.get_template(action_name) else 'unknown'}

Please output a JSON object with a single key "duration" (in seconds, a float). Use a reasonable estimate based on the parent duration and typical biomechanics.
If you cannot estimate, output {{"duration": null}}.
"""
        try:
            response = await self.client.chat.completions.create(
                model="deepseek-v4-flash",
                messages=[
                    {"role": "system", "content": "You are an expert in biomechanical and physical action timing."},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"},
                temperature=0.3,
                max_tokens=50
            )
            data = json.loads(response.choices[0].message.content)
            estimated_duration = data.get("duration")
            if estimated_duration is None:
                template = action_storage.get_template(action_name)
                if template:
                    estimated_duration = template.default_duration
                else:
                    return None
        except Exception as e:
            print(f"LLM instantiation failed for {action_name}: {e}")
            template = action_storage.get_template(action_name)
            if template:
                estimated_duration = template.default_duration
            else:
                return None

        participants = []
        for p_name in ancestor_participants:
            obj = memory.objects.get(p_name)
            if obj:
                participants.append(obj)
        if not participants:
            participants = [p for p in memory.objects.values() if 'hand' in p.name or 'arm' in p.name]

        sub_action_desc = {
            "name": action_name,
            "participants": [p.name for p in participants],
            "tags": [action_name, "estimated"]
        }
        memory.extensions.append({
            "parent": ancestor,
            "sub_action": sub_action_desc,
            "estimated_duration": estimated_duration
        })
        memory.apply_extensions(action_storage, object_storage)

        return memory._find_action_by_name(action_name)

# Memory Ingestor

class MemoryIngestor:
    def __init__(self, object_storage: ObjectStorage, action_storage: ActionStorage, search_engine: SearchEngine):
        self.object_storage = object_storage
        self.action_storage = action_storage
        self.se = search_engine
        self.decomposer = ActionDecomposer() if os.getenv("DEEPSEEK_API_KEY") else None

    async def ingest(self, json_data: dict) -> Memory:
        person = Protagonist(json_data["protagonist"]["name"])
        mem = Memory(json_data["memory_id"], person)

        for obj_data in json_data.get("objects", []):
            name = obj_data["name"]
            overrides = obj_data.get("overrides", {})
            instance = self.object_storage.instantiate(name, overrides)
            for rel in obj_data.get("spatial_relations", []):
                target = mem.objects.get(rel["target"])
                if target:
                    instance.add_spatial(rel["relation"], target)
            mem.add_object(instance)

        actions_to_add = []
        for act_data in json_data["actions"]:
            action = self.action_storage.instantiate(act_data["template"], act_data, auto_add=True)

            def resolve_participants(action_obj: Action):
                resolved = []
                for p_name in action_obj._participant_names:
                    if p_name in mem.objects:
                        resolved.append(mem.objects[p_name])
                    else:
                        raise ValueError(f"Participant '{p_name}' not found.")
                action_obj.participants = resolved
                for sub in action_obj.sub_actions:
                    resolve_participants(sub)

            resolve_participants(action)

            def convert_changes(action_obj: Action):
                new_changes = []
                for change in action_obj.changes:
                    if isinstance(change, dict):
                        obj = mem.objects.get(change["object"])
                        if obj is None:
                            raise ValueError(f"Object '{change['object']}' not found.")
                        new_changes.append((obj, change["attribute"], change.get("old"), change["new"]))
                    else:
                        new_changes.append(change)
                action_obj.changes = new_changes

                new_physical = []
                for phys in action_obj.physical_changes:
                    if isinstance(phys, dict):
                        obj = mem.objects.get(phys["object"])
                        if obj is None:
                            raise ValueError(f"Object '{phys['object']}' not found.")
                        new_physical.append((obj, phys["attribute"], phys.get("old"), phys["new"]))
                    else:
                        new_physical.append(phys)
                action_obj.physical_changes = new_physical

                new_preconditions = []
                for prec in action_obj.preconditions:
                    if isinstance(prec, dict):
                        obj = mem.objects.get(prec["object"])
                        if obj is None:
                            raise ValueError(f"Object '{prec['object']}' not found.")
                        new_preconditions.append((obj, prec["attribute"], prec["value"]))
                    else:
                        new_preconditions.append(prec)
                action_obj.preconditions = new_preconditions

                for sub in action_obj.sub_actions:
                    convert_changes(sub)

            convert_changes(action)
            actions_to_add.append(action)

        if self.decomposer:
            for action in actions_to_add:
                decomposed_action = await self.decomposer.decompose_action_tree(action, self.action_storage, self.object_storage)

        current_time = 0.0
        for action in actions_to_add:
            total_duration = self.action_storage.compute_start_offsets(action, current_time)
            current_time += total_duration

        for action in actions_to_add:
            mem.add_action(action)

        self.se.add_memory(mem)
        return mem

We will adopt cyclical encoding for continuous, repetitive actions. The goal is to keep the action tree compact while still capturing the essence of a repeated motion (like rotating an apple for 40 seconds) without exploding it into dozens of identical micro‑actions.


1. What Cyclical Encoding Means
A cyclical action is an atomic template that models a single cycle of a repetitive movement. When instantiated, it is given a repetitions (or cycles) parameter. The total duration of the action equals cycle_duration × repetitions, and the action is treated as a single unbroken event for planning and search.
Example: cyclic_finger_rotation
Participants: [hand, object]
cycle_duration = 1.0 (one complete rotation assist cycle)
kinematic_trajectory = "cyclical_shear"
repetitions = 36 → total duration 36 s
The action appears in the memory exactly once, with all the relevant metadata. The internal biomechanical details (flex‑extend‑flex…) are not stored inside this memory — they live only in the action registry as a future decomposition option.


2. Modifications to the Action System
We extend ActionTemplate and Action to support cyclical behaviour.
2.1 New Template Fields
python
@dataclass
class ActionTemplate:
    # ... existing fields ...
    is_cyclical: bool = False
    default_cycle_duration: float = 0.0          # seconds per cycle
    max_cycles: Optional[int] = None             # optional upper bound
    # for a cyclical action, the total default_duration may be ignored;
    # the actual duration is computed from overrides.
2.2 Extended Action Fields
python
@dataclass
class Action:
    # ... existing fields ...
    repetitions: int = 1                         # for cyclical actions
    cycle_duration: float = 0.0                  # seconds per repetition

    @property
    def effective_duration(self) -> float:
        if self.temporal_type == "cyclical":
            return self.cycle_duration * self.repetitions
        return self.duration

        return self._resolve_required_attrs(name)

    def instantiate(self, name: str, overrides: Dict[str, Any] = None) -> Object:
        if overrides:
            if name not in self._defaults:
                self._defaults[name] = {}
            for key in overrides.keys():
                if key not in self._defaults[name]:
                    self._defaults[name][key] = overrides[key]
        required = self._resolve_required_attrs(name)
        if overrides:
            missing = required - set(overrides.keys())
            if missing:
                raise ValueError(f"Missing required attributes for '{name}': {missing}")
        else:
            if required:
                raise ValueError(f"Required attributes for '{name}' not provided: {required}")
        instance = Object(name, template_name=name)
        if overrides:
            instance.overrides = overrides.copy()
        return instance

    def list_templates(self) -> List[str]:
        return list(self._defaults.keys())

class ActionStorage:
    _instance = None
    _templates: Dict[str, ActionTemplate] = {}
    _name_embeddings: Dict[str, np.ndarray] = {}
    _embedding_model = None
    _flat_names_cache: Dict[str, List[str]] = {}
    # reverse index: template name -> set of memory ids (or Memory objects)
    _usage_index: Dict[str, Set['Memory']] = defaultdict(set)

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            if SentenceTransformer:
                cls._embedding_model = SentenceTransformer("all-MiniLM-L6-v2")
        return cls._instance

    def add_template(self, template: ActionTemplate):
        old_template = self._templates.get(template.name)
        was_atomic = old_template is not None and not old_template.sub_action_templates
        became_complex = bool(template.sub_action_templates)

        self._templates[template.name] = template
        if template.name in self._flat_names_cache:
            del self._flat_names_cache[template.name]

        # Propagation of new decomposition to old memories
        if was_atomic and became_complex:
            self._propagate_new_decomposition(template)

    def _propagate_new_decomposition(self, template: ActionTemplate):
        """Add decomposition_reference extensions to all memories that used the old atomic version."""
        affected_memories = self._usage_index.get(template.name, set())
        for memory in affected_memories:
            # Ensure the parent action exists in the memory
            parent_node = memory._find_action_by_name(template.name)
            if parent_node is None:
                continue
            # Build a lightweight reference extension
            new_sub_names = [sub_name for sub_name in template.sub_action_templates]
            extension = {
                "type": "decomposition_reference",
                "parent": template.name,
                "new_sub_actions": [{"template_name": sn} for sn in new_sub_names]
            }
            memory.extensions.append(extension)

    def register_memory_usage(self, template_name: str, memory: 'Memory'):
        """Called during ingestion after deep decomposition for every template used."""
        self._usage_index[template_name].add(memory)

    def unregister_memory(self, memory: 'Memory'):
        for mem_set in self._usage_index.values():
            mem_set.discard(memory)

    def get_memories_using(self, template_name: str) -> Set['Memory']:
        return self._usage_index.get(template_name, set())

    def get_template(self, name: str) -> Optional[ActionTemplate]:
        return self._templates.get(name)

    def instantiate(self, name: str, overrides: Dict[str, Any], auto_add: bool = True) -> Action:
        template = self._templates.get(name)
        if template is None:
            if auto_add:
                participants = overrides.get("participants", ["hand", "object"])
                cat = overrides.get("action_category", "INTERACTION" if len(participants) > 1 else "BODY_ONLY")
                is_cyclical = overrides.get("temporal_type") == "cyclical"
                new_template = ActionTemplate(
                    name=name,
                    default_duration=overrides.get("duration", 1),
                    action_category=cat,
                    default_participants=participants,
                    required_overrides=["changes"],
                    is_cyclical=is_cyclical,
                    default_cycle_duration=overrides.get("cycle_duration", 0.0)
                )
                self.add_template(new_template)
                template = new_template
                print(f"✅ Auto‑added new action: {name} ({cat})")
            else:
                raise ValueError(f"Unknown action template: {name}")

        missing = set(template.required_overrides) - set(overrides.keys())
        if missing:
            raise ValueError(f"Missing required overrides for action '{name}': {missing}")

        provided_participants = overrides.get("participants", [])
        template_roles = template.default_participants
        if provided_participants:
            if len(provided_participants) != len(template_roles):
                raise ValueError(f"Participant mismatch: need {len(template_roles)}, got {len(provided_participants)}")
            binding = dict(zip(template_roles, provided_participants))
        else:
            binding = {}

        bound_kinematics = copy.deepcopy(template.kinematic_metadata)
        bound_kinematics["bound_target"] = binding.get("target_object", None)
        bound_kinematics["bound_hand"] = binding.get("hand", binding.get("arm", None))

        if template.is_cyclical:
            reps = overrides.get("repetitions", 1)
            cycle_dur = overrides.get("cycle_duration", template.default_cycle_duration)
            total_duration = int(round(cycle_dur * reps))
        else:
            reps = 1
            cycle_dur = 0.0
            total_duration = overrides.get("duration", template.default_duration)

        final_tags = overrides.get("tags", template.default_tags)
        final_physical_changes = template.default_physical_changes
        final_physical_relations = template.default_physical_relations
        final_changes = overrides.get("changes", [])
        final_spatial_changes = overrides.get("spatial_changes", [])
        final_preconditions = template.preconditions

        # Build sub-actions (only if not cyclical, or if cyclical with base template)
        sub_actions = []
        if not template.is_cyclical:
            for sub_name in template.sub_action_templates:
                sub_defaults = template.sub_action_defaults.get(sub_name, {})
                sub_override = overrides.get("sub_actions", {}).get(sub_name, {})
                merged_sub = {**sub_defaults, **sub_override}
                sub_action = self.instantiate(sub_name, merged_sub, auto_add=False)
                sub_actions.append(sub_action)

        action = Action(
            name=name,
            duration=total_duration,
            participants=[],
            changes=final_changes,
            physical_changes=final_physical_changes,
            physical_relations=final_physical_relations,
            tags=final_tags,
            action_category=template.action_category,
            kinematic_metadata=bound_kinematics,
            kinematic_trajectory=template.kinematic_trajectory,
            preconditions=final_preconditions,
            sub_actions=sub_actions,
            temporal_type=template.temporal_type if not template.is_cyclical else "cyclical",
            _participant_names=provided_participants,
            repetitions=reps,
            cycle_duration=cycle_dur
        )
        return action

    def list_templates(self) -> List[str]:
        return list(self._templates.keys())

    def _get_embedding(self, text: str) -> np.ndarray:
        if self._embedding_model is None:
            return np.zeros(384)
        if text not in self._name_embeddings:
            self._name_embeddings[text] = self._embedding_model.encode(text, normalize_embeddings=True)
        return self._name_embeddings[text]

    def _similarity_between_names(self, name1: str, name2: str) -> float:
        if name1 == name2:
            return 1.0
        emb1 = self._get_embedding(name1)
        emb2 = self._get_embedding(name2)
        return float(np.dot(emb1, emb2))

    def _substitution_cost(self, name1: str, name2: str) -> float:
        if name1 == name2:
            return 0.0
        if self._embedding_model is None:
            return 1.0
        sim = self._similarity_between_names(name1, name2)
        if sim > 0.9:
            return 0.1
        elif sim > 0.8:
            return 0.3
        else:
            return 1.0

    def _weighted_edit_distance(self, seq1: List[str], seq2: List[str]) -> float:
        n, m = len(seq1), len(seq2)
        dp = [[0.0] * (m + 1) for _ in range(n + 1)]
        for i in range(n + 1):
            dp[i][0] = float(i)
        for j in range(m + 1):
            dp[0][j] = float(j)
        for i in range(1, n + 1):
            for j in range(1, m + 1):
                cost = self._substitution_cost(seq1[i-1], seq2[j-1])
                dp[i][j] = min(
                    dp[i-1][j] + 1.0,
                    dp[i][j-1] + 1.0,
                    dp[i-1][j-1] + cost
                )
        distance = dp[n][m]
        max_len = max(n, m)
        if max_len == 0:
            return 1.0
        return 1.0 - (distance / max_len)

    def _flatten_template_names(self, template: ActionTemplate) -> List[str]:
        if template.name in self._flat_names_cache:
            return self._flat_names_cache[template.name]
        if not template.sub_action_templates:
            result = [template.name]
        else:
            result = []
            for sub_name in template.sub_action_templates:
                sub_template = self._templates.get(sub_name)
                if sub_template:
                    result.extend(self._flatten_template_names(sub_template))
        self._flat_names_cache[template.name] = result
        return result

    def _flatten_action_names(self, action: Action) -> List[str]:
        if not action.sub_actions:
            return [action.name]
        result = []
        for sub in action.sub_actions:
            result.extend(self._flatten_action_names(sub))
        return result

    def get_parents(self, action_name: str) -> List[str]:
        if not hasattr(self, '_parent_map'):
            self._parent_map = self._build_parent_map()
        return self._parent_map.get(action_name, [])

    def _build_parent_map(self):
        parent_map = defaultdict(list)
        for name, template in self._templates.items():
            for sub in template.sub_action_templates:
                parent_map[sub].append(name)
        return parent_map

    def find_similar_atomic_action(self, action: Action) -> Optional[Action]:
        if self._embedding_model is None:
            return None

        proposed = {
            "name": action.name,
            "participants": [p.name for p in action.participants],
            "trajectory": action.kinematic_trajectory,
            "category": action.action_category
        }

        best_match = None
        best_score = -1.0
        for name, template in self._templates.items():
            if template.sub_action_templates:
                continue
            score = self._compute_atomic_similarity(proposed, template)
            if score > best_score:
                best_score = score
                best_match = name

        if best_match and best_score > 0.0:
            return self.instantiate(best_match, {"changes": action.changes}, auto_add=False)
        return None

    def _compute_atomic_similarity(self, proposed: Dict, template: ActionTemplate) -> float:
        if self._similarity_between_names(proposed["name"], template.name) < 0.85:
            return 0.0
        if self._similarity_between_names(proposed.get("trajectory", "straight"), template.kinematic_trajectory) < 0.96:
            return 0.0
        if proposed.get("category", "INTERACTION") != template.action_category:
            return 0.0
        p1 = proposed.get("participants", [])
        p2 = template.default_participants
        if len(p1) != len(p2):
            return 0.0
        for a, b in zip(p1, p2):
            if self._similarity_between_names(a, b) < 0.96:
                return 0.0
        return 1.0

    def find_complex_template_by_flat_names(self, flat_names: List[str], threshold: float = 1.0) -> Optional[str]:
        """Exact match by default (threshold 1.0). Use >0.75 for approximate."""
        if not flat_names:
            return None
        best_match = None
        best_score = -1.0

        for name, template in self._templates.items():
            if not template.sub_action_templates:
                continue
            template_flat = self._flatten_template_names(template)
            sim = self._weighted_edit_distance(flat_names, template_flat)
            if sim > best_score:
                best_score = sim
                best_match = name

        if best_match and best_score >= threshold:
            print(f"🔍 Found similar complex action: '{best_match}' (score: {best_score:.3f})")
            return best_match
        return None

    def compute_start_offsets(self, action: Action, parent_start: float = 0.0) -> float:
        action.start_offset = parent_start
        if not action.sub_actions:
            dur = action.effective_duration
            return dur

        if action.temporal_type == "sequential":
            current = parent_start
            for sub in action.sub_actions:
                child_dur = self.compute_start_offsets(sub, current)
                current += child_dur
            total_duration = current - parent_start
        elif action.temporal_type == "parallel":
            max_end = parent_start
            for sub in action.sub_actions:
                child_dur = self.compute_start_offsets(sub, parent_start)
                max_end = max(max_end, parent_start + child_dur)
            total_duration = max_end - parent_start
        elif action.temporal_type == "cyclical":
            total_duration = action.cycle_duration * action.repetitions
        else:
            total_duration = action.duration

        action.duration = int(round(total_duration))
        return total_duration

class Memory:
    def __init__(self, memory_id: str, protagonist: Protagonist):
        self.id = memory_id
        self.protagonist = protagonist
        self.objects: Dict[str, Object] = {protagonist.name: protagonist}
        self.actions: List[Action] = []
        self.timeline: List[int] = []
        # Extensions for lazy sub‑action addition
        self.extensions: List[Dict] = []
        self.extensions_applied: bool = False
        # referenced sub-actions from decomposition propagation (not yet instantiated)
        self._referenced_sub_actions: Dict[str, Set[str]] = defaultdict(set)  # parent -> set of sub template names

    def add_object(self, obj: Object):
        self.objects[obj.name] = obj

    def add_action(self, action: Action):
        self.actions.append(action)
        self._apply_action(action)
        if self.timeline:
            self.timeline.append(self.timeline[-1] + action.effective_duration)
        else:
            self.timeline.append(action.effective_duration)

    def _apply_action(self, action: Action):
        for obj, attr, old, new in action.changes:
            if obj.get_attr(attr) == old or old is None:
                obj.overrides[attr] = new
        for obj1_name, rel, obj2_name, op in action.spatial_changes:
            if obj1_name in self.objects and obj2_name in self.objects:
                obj1 = self.objects[obj1_name]
                obj2 = self.objects[obj2_name]
                if op == "add":
                    if (obj1_name, rel, obj2_name) not in obj1.spatial_relations:
                        obj1.spatial_relations.append((obj1_name, rel, obj2_name))
                elif op == "remove":
                    obj1.spatial_relations = [r for r in obj1.spatial_relations
                                             if not (r[0]==obj1_name and r[1]==rel and r[2]==obj2_name)]
        for sub in action.sub_actions:
            self._apply_action(sub)

    def _find_action_by_name(self, name: str) -> Optional[Action]:
        for act in self.actions:
            found = self._search_action(act, name)
            if found:
                return found
        return None

    def _search_action(self, action: Action, name: str) -> Optional[Action]:
        if action.name == name:
            return action
        for sub in action.sub_actions:
            found = self._search_action(sub, name)
            if found:
                return found
        return None

    def apply_extensions(self, action_storage: 'ActionStorage', object_storage: 'ObjectStorage'):
        for ext in self.extensions:
            if ext.get("type") == "decomposition_reference":
                # Just record the referenced sub-actions for indexing
                parent_name = ext["parent"]
                new_subs = [s["template_name"] for s in ext["new_sub_actions"]]
                self._referenced_sub_actions[parent_name].update(new_subs)
            else:
                # Existing instantiation extension
                parent_name = ext["parent"]
                sub_action_data = ext["sub_action"]
                estimated_duration = ext.get("estimated_duration")

                parent_action = self._find_action_by_name(parent_name)
                if parent_action is None:
                    continue

                if any(sub.name == sub_action_data["name"] for sub in parent_action.sub_actions):
                    continue

                template = action_storage.get_template(sub_action_data["name"])
                if template is None:
                    continue

                participants = []
                for p_name in sub_action_data.get("participants", []):
                    obj = self.objects.get(p_name)
                    if obj is None:
                        obj = Object(p_name)
                        self.objects[p_name] = obj
                    participants.append(obj)

                overrides = {
                    "participants": [p.name for p in participants],
                    "duration": int(round(estimated_duration)) if estimated_duration else template.default_duration,
                    "tags": sub_action_data.get("tags", [])
                }
                sub_action = action_storage.instantiate(sub_action_data["name"], overrides, auto_add=False)
                parent_action.sub_actions.append(sub_action)

        self.extensions_applied = True

    def ensure_extensions_applied(self, action_storage: 'ActionStorage', object_storage: 'ObjectStorage'):
        if not self.extensions_applied:
            self.apply_extensions(action_storage, object_storage)
            self.extensions_applied = True

    def export_initial_world_state(self) -> 'WorldState':
        ws = WorldState()
        for obj in self.objects.values():
            ws.add_object_state(obj.id, obj.attributes, obj.name)
            for rel in obj.spatial_relations:
                other = self.objects.get(rel[2])
                if other:
                    ws.add_spatial(obj.id, rel[1], other.id)
        return ws

    }

    def __init__(self, objects: Optional[Dict[str, Dict[str, Any]]] = None,

        self.spatial_reverse: Dict[str, Dict[str, Set[str]]] = defaultdict(lambda: defaultdict(set))

    def add_spatial(self, subj_id: str, rel: str, obj_id: str):
        self.spatial_forward[subj_id][rel].add(obj_id)
        self.spatial_reverse[obj_id][rel].add(subj_id)

    # (other WorldState methods remain unchanged)
    def remove_spatial(self, subj_id: str, rel: str, obj_id: str):
        if subj_id in self.spatial_forward and rel in self.spatial_forward[subj_id]:
            self.spatial_forward[subj_id][rel].discard(obj_id)
        if obj_id in self.spatial_reverse and rel in self.spatial_reverse[obj_id]:
            self.spatial_reverse[obj_id][rel].discard(subj_id)

    def get_direct_related(self, obj_id: str, rel: str, reverse: bool = False) -> Set[str]:
        if reverse:
            return self.spatial_reverse.get(obj_id, {}).get(rel, set())
        else:
            return self.spatial_forward.get(obj_id, {}).get(rel, set())

    def _dfs(self, start: str, rel: str, forward: bool) -> Set[str]:
        visited = set()
        result = set()
        stack = [start]
        while stack:
            current = stack.pop()
            if current in visited:
                continue
            visited.add(current)
            if forward:
                neighbours = self.spatial_forward.get(current, {}).get(rel, set())
            else:
                neighbours = self.spatial_reverse.get(current, {}).get(rel, set())
            for nbr in neighbours:
                if nbr not in visited:
                    result.add(nbr)
                    stack.append(nbr)
        return result

    def get_transitive_descendants(self, subj_id: str, rel: str) -> Set[str]:
        if rel not in self.TRANSITIVE_RELATIONS:
            return self.get_direct_related(subj_id, rel, reverse=False)
        return self._dfs(subj_id, rel, forward=True)

  "base_action_name": "cyclic_finger_rotation",
  "cycle_duration": 1.0,
  "repetitions": 36,
  "sub_actions": []
}
The base_action will be further decomposed recursively.

If no decomposition, return {"sub_actions": []}.
"""

    async def deep_decompose(
        self,
        action: Action,
        action_storage: ActionStorage,
        object_storage: ObjectStorage,
        object_pool: Dict[str, Object]
    ) -> Action:
        """Recursively decompose an action tree until all leaves are atomic or cyclical.
           Returns the (possibly replaced) action with a fully expanded subtree."""
        # First, recursively decompose children
        if action.sub_actions:
            decomposed_children = []
            for sub in action.sub_actions:
                decomposed_sub = await self.deep_decompose(sub, action_storage, object_storage, object_pool)
                decomposed_children.append(decomposed_sub)
            action.sub_actions = decomposed_children
            # After children are resolved, try to match/register as complex action
            return await self._handle_complex_action(action, action_storage, object_storage, object_pool)

        # Leaf node – try atomic match or decompose
        return await self._handle_leaf_action(action, action_storage, object_storage, object_pool)

    async def _handle_leaf_action(self, action, action_storage, object_storage, object_pool):
        # 1. Check if a matching atomic template already exists
        matched = action_storage.find_similar_atomic_action(action)
        if matched:
            # Re‑instantiate with the matched template, but keep the original's concrete participants & context
            return self._merge_into_matched(action, matched, object_pool)

        # 2. Ask LLM
        sub_data = await self._call_llm_for_decomposition(action)
        if sub_data is None or len(sub_data) == 0:
            # LLM considers this atomic – register as new atomic
            self._register_atomic_action(action, action_storage)
            return action

        # 3. Build sub-actions from LLM output
        instantiated_subs = []
        for sub_d in sub_data:
            if sub_d.get("decomposition_type") == "cyclical":
                # Create cyclical action, then recursively decompose its base action
                cyclical_action = self._build_cyclical_action(sub_d, action.participants, object_pool)
                cyclical_action = await self.deep_decompose(cyclical_action, action_storage, object_storage, object_pool)
                instantiated_subs.append(cyclical_action)
            else:
                sub_action = self._build_action_from_dict(sub_d, action.participants, object_pool)
                sub_action = await self.deep_decompose(sub_action, action_storage, object_storage, object_pool)
                instantiated_subs.append(sub_action)

        # Replace current leaf with a complex action holding these children
        action.sub_actions = instantiated_subs
        action.temporal_type = "sequential"   # could be overridden by LLM later
        return await self._handle_complex_action(action, action_storage, object_storage, object_pool)

    async def _handle_complex_action(self, action, action_storage, object_storage, object_pool):
        flat_names = action_storage._flatten_action_names(action)
        # exact match by default (threshold 1.0)
        matched_name = action_storage.find_complex_template_by_flat_names(flat_names, threshold=1.0)
        if matched_name:
            # Replace with matched template instance (keeping original context)
            return self._instantiate_from_template(matched_name, action, object_pool)
        else:
            # Register as new complex action
            self._register_complex_action(action, action_storage)
            return action

    def _build_cyclical_action(self, data: Dict, parent_participants: List[Object], object_pool: Dict[str,Object]) -> Action:
        base_name = data["base_action_name"]
        cycle_dur = data["cycle_duration"]
        reps = data["repetitions"]
        # Create a dummy leaf action for the cycle body and recursively decompose it
        body_action = Action(
            name=base_name,
            duration=int(cycle_dur),
            participants=parent_participants,  # for now; will be resolved
            changes=[],
            physical_changes=[],
            physical_relations=[],
            tags=[base_name],
            action_category="INTERACTION",
            kinematic_metadata={},
            kinematic_trajectory="cyclical",
            temporal_type="atomic",
            _participant_names=[p.name for p in parent_participants]
        )
        # Return a cyclical wrapper; deep_decompose will replace the body later
        wrapper = Action(
            name=data.get("action_name", "cyclic_action"),
            duration=int(cycle_dur * reps),
            participants=parent_participants,
            changes=[],
            physical_changes=[],
            physical_relations=[],
            tags=[base_name],
            action_category="INTERACTION",
            kinematic_metadata={},
            kinematic_trajectory="cyclical",
            temporal_type="cyclical",
            repetitions=reps,
            cycle_duration=cycle_dur,
            _participant_names=[p.name for p in parent_participants],
            sub_actions=[body_action]
        )
        return wrapper

    def _build_action_from_dict(self, d: Dict, parent_participants: List[Object], object_pool: Dict[str,Object]) -> Action:
        # Build a leaf action with the given data (participants resolved from object_pool)
        resolved_parts = self._resolve_participants(d.get("participants", []), parent_participants, object_pool)
        return Action(
            name=d["name"],
            duration=d.get("duration", 1),
            participants=resolved_parts,
            changes=[],
            physical_changes=[],
            physical_relations=[],
            tags=[d["name"]],
            action_category="INTERACTION",
            kinematic_metadata=d.get("kinematic_metadata", {}),
            kinematic_trajectory=d.get("kinematic_trajectory", "straight"),
            temporal_type=d.get("temporal_type", "sequential"),
            _participant_names=[p.name for p in resolved_parts]
        )

    def _resolve_participants(self, llm_names: List[str], parent_participants: List[Object], object_pool: Dict[str,Object]) -> List[Object]:
        resolved = []
        for llm_name in llm_names:
            # try exact match in object_pool (memory objects)
            if llm_name in object_pool:
                resolved.append(object_pool[llm_name])
                continue
            # try matching by name in parent participants
            matched = None
            for p in parent_participants:
                if p.name == llm_name:
                    matched = p
                    break
            if matched:
                resolved.append(matched)
            else:
                # create a new object placeholder
                new_obj = Object(llm_name)
                object_pool[llm_name] = new_obj
                resolved.append(new_obj)
        return resolved

    def _merge_into_matched(self, original: Action, matched: Action, object_pool) -> Action:
        # Use matched template but keep original participants and context
        matched.participants = original.participants
        matched._participant_names = original._participant_names
        matched.changes = original.changes
        matched.tags = list(set(original.tags + matched.tags))
        return matched

    def _register_atomic_action(self, action: Action, action_storage: ActionStorage):
        template = ActionTemplate(
            name=action.name,
            default_duration=action.duration,
            action_category=action.action_category,
            default_participants=[p.name for p in action.participants],
            default_tags=action.tags,
            kinematic_metadata=action.kinematic_metadata,
            kinematic_trajectory=action.kinematic_trajectory,
            preconditions=action.preconditions,
            required_overrides=["changes"],
            temporal_type="atomic"
        )
        action_storage.add_template(template)

    def _register_complex_action(self, action: Action, action_storage: ActionStorage):
        sub_names = []
        sub_defaults = {}
        for sub in action.sub_actions:
            sub_names.append(sub.name)
            sub_defaults[sub.name] = {
                "participants": [p.name for p in sub.participants],
                "changes": sub.changes,
                "duration": sub.duration,
                "tags": sub.tags,
                "kinematic_trajectory": sub.kinematic_trajectory
            }
        template = ActionTemplate(
            name=action.name,
            default_duration=action.duration,
            action_category=action.action_category,
            default_participants=[p.name for p in action.participants],
            default_tags=action.tags,
            kinematic_metadata=action.kinematic_metadata,
            kinematic_trajectory=action.kinematic_trajectory,
            preconditions=action.preconditions,
            required_overrides=["changes"],
            sub_action_templates=sub_names,
            sub_action_defaults=sub_defaults,
            temporal_type=action.temporal_type
        )
        action_storage.add_template(template)

    async def _call_llm_for_decomposition(self, action: Action) -> List[Dict]:
        participant_names = [p.name for p in action.participants]
        prompt = f"""
Action name: {action.name}
Participants: {participant_names}
Kinematic metadata: {action.kinematic_metadata}
Current physical changes: {action.changes}
Tags: {action.tags}

{self.default_instructions}
"""
        try:
            response = await self.client.chat.completions.create(
                model="deepseek-v4-flash",
                messages=[
                    {"role": "system", "content": "You are an expert in physical, mechanical, and biomechanical task decomposition."},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"},
                temperature=0.3,
                max_tokens=500
            )
            data = json.loads(response.choices[0].message.content)
            return data.get("sub_actions", [])
        except Exception as e:
            print(f"LLM decomposition failed for {action.name}: {e}")
            return []

    async def instantiate_reference(
        self,
        memory: Memory,
        parent_name: str,
        sub_action_name: str,
        object_storage: ObjectStorage,
        action_storage: ActionStorage
    ) -> Optional[Action]:
        """
        Lazily instantiate a referenced sub-action with context from the parent.
        This is called during search refinement.
        """
        parent_action = memory._find_action_by_name(parent_name)
        if parent_action is None:
            return None

        # Use LLM to estimate duration based on parent context
        memory_text = f"Memory ID: {memory.id}\n"
        for act in memory.actions:
            memory_text += f"Action: {act.name} (duration {act.duration}s)\n"
        prompt = f"""
You are an expert in estimating the duration of a sub-action based on context.
We have a memory that contains the action '{parent_name}' which took {parent_action.duration} seconds.
We want to estimate the duration of the sub‑action '{sub_action_name}', which is part of '{parent_name}'.

Memory context:
{memory_text}

Participants of '{parent_name}': {[p.name for p in parent_action.participants]}
Template for '{sub_action_name}': {action_storage.get_template(sub_action_name).kinematic_metadata if action_storage.get_template(sub_action_name) else 'unknown'}

Please output a JSON object with a single key "duration" (in seconds, a float). Use a reasonable estimate based on the parent duration and typical biomechanics.
If you cannot estimate, output {{"duration": null}}.
"""
        try:
            response = await self.client.chat.completions.create(
                model="deepseek-v4-flash",
                messages=[
                    {"role": "system", "content": "You are an expert in biomechanical and physical action timing."},
                    {"role": "user", "content": prompt}
                ],
                response_format={"type": "json_object"},
                temperature=0.3,
                max_tokens=50
            )
            data = json.loads(response.choices[0].message.content)
            estimated_duration = data.get("duration")
        except Exception as e:
            print(f"LLM instantiation failed for {sub_action_name}: {e}")
            estimated_duration = None

        if estimated_duration is None:
            template = action_storage.get_template(sub_action_name)
            if template:
                estimated_duration = template.default_duration
            else:
                return None

        # Build the sub-action
        sub_template = action_storage.get_template(sub_action_name)
        if not sub_template:
            return None

        # Resolve participants from parent
        participants = []
        for p_name in sub_template.default_participants:
            obj = memory.objects.get(p_name)
            if not obj:
                obj = Object(p_name)
                memory.objects[p_name] = obj
            participants.append(obj)

        overrides = {
            "participants": [p.name for p in participants],
            "duration": int(round(estimated_duration)),
            "tags": [sub_action_name, "estimated"]
        }
        sub_action = action_storage.instantiate(sub_action_name, overrides, auto_add=False)
        parent_action.sub_actions.append(sub_action)
        return sub_action

# Memory Pipeline (ingestion + deep decomposition + usage registration)
# ----------------------------------------------------------------------
class MemoryPipeline:
    def __init__(self, ingestor: 'MemoryIngestor', decomposer: ActionDecomposer):
        self.ingestor = ingestor
        self.decomposer = decomposer

    async def ingest_and_decompose(self, json_data: dict) -> Memory:
        # 1. Regular ingestion (builds initial tree)
        memory = await self.ingestor.ingest(json_data)

        # 2. Deep‑decompose every top‑level action
        for i, action in enumerate(memory.actions):
            memory.actions[i] = await self.decomposer.deep_decompose(
                action,
                self.ingestor.action_storage,
                self.ingestor.object_storage,
                memory.objects
            )

        # 3. Recompute timeline
        current_time = 0.0
        for action in memory.actions:
            total_duration = self.ingestor.action_storage.compute_start_offsets(action, current_time)
            current_time += total_duration

        # 4. Register usage of all templates (leaf and complex)
        def register_usage(action: Action, mem: Memory):
            if action.sub_actions:
                # complex action -> register the complex template name
                self.ingestor.action_storage.register_memory_usage(action.name, mem)
                for sub in action.sub_actions:
                    register_usage(sub, mem)
            else:
                # leaf action -> register its template name
                self.ingestor.action_storage.register_memory_usage(action.name, mem)

        for action in memory.actions:
            register_usage(action, memory)

        # 5. Add to search engine (the ingestor already added it, but we re-add after decomposition?)
        # The ingestor already adds to search engine; we just need to re-index with the updated tree.
        # Re-index by removing and re-adding is heavy; we can update index in place. For simplicity,
        # the SearchEngine.add_memory can be called again (it handles re-indexing by overwriting).
        # We'll assume the search engine's add_memory is idempotent w.r.t. to re-indexing the same memory.
        self.ingestor.se.add_memory(memory)   # re-index with final tree

        return memory

# Memory Ingestor (unchanged, but uses pipeline now)

class MemoryIngestor:
    def __init__(self, object_storage: ObjectStorage, action_storage: ActionStorage, search_engine: 'SearchEngine'):
        self.object_storage = object_storage
        self.action_storage = action_storage
        self.se = search_engine

    async def ingest(self, json_data: dict) -> Memory:
        person = Protagonist(json_data["protagonist"]["name"])
        mem = Memory(json_data["memory_id"], person)

        for obj_data in json_data.get("objects", []):
            name = obj_data["name"]
            overrides = obj_data.get("overrides", {})
            instance = self.object_storage.instantiate(name, overrides)
            for rel in obj_data.get("spatial_relations", []):
                target = mem.objects.get(rel["target"])
                if target:
                    instance.add_spatial(rel["relation"], target)
            mem.add_object(instance)

        actions_to_add = []
        for act_data in json_data["actions"]:
            action = self.action_storage.instantiate(act_data["template"], act_data, auto_add=True)

            def resolve_participants(action_obj: Action):
                resolved = []
                for p_name in action_obj._participant_names:
                    if p_name in mem.objects:
                        resolved.append(mem.objects[p_name])
                    else:
                        raise ValueError(f"Participant '{p_name}' not found.")
                action_obj.participants = resolved
                for sub in action_obj.sub_actions:
                    resolve_participants(sub)

            resolve_participants(action)

            def convert_changes(action_obj: Action):
                new_changes = []
                for change in action_obj.changes:
                    if isinstance(change, dict):
                        obj = mem.objects.get(change["object"])
                        if obj is None:
                            raise ValueError(f"Object '{change['object']}' not found.")
                        new_changes.append((obj, change["attribute"], change.get("old"), change["new"]))
                    else:
                        new_changes.append(change)
                action_obj.changes = new_changes

                new_physical = []
                for phys in action_obj.physical_changes:
                    if isinstance(phys, dict):
                        obj = mem.objects.get(phys["object"])
                        if obj is None:
                            raise ValueError(f"Object '{phys['object']}' not found.")
                        new_physical.append((obj, phys["attribute"], phys.get("old"), phys["new"]))
                    else:
                        new_physical.append(phys)
                action_obj.physical_changes = new_physical

                new_preconditions = []
                for prec in action_obj.preconditions:
                    if isinstance(prec, dict):
                        obj = mem.objects.get(prec["object"])
                        if obj is None:
                            raise ValueError(f"Object '{prec['object']}' not found.")
                        new_preconditions.append((obj, prec["attribute"], prec["value"]))
                    else:
                        new_preconditions.append(prec)
                action_obj.preconditions = new_preconditions

                for sub in action_obj.sub_actions:
                    convert_changes(sub)

            convert_changes(action)
            actions_to_add.append(action)

        # No decomposition here – that will be done by MemoryPipeline
        for action in actions_to_add:
            mem.add_action(action)

        self.se.add_memory(mem)   # index initial tree (will be re-indexed later)
        return mem

class SearchEngine:
    def __init__(self, max_memories: int = 10, use_hybrid: bool = False,
                 action_storage: ActionStorage = None, object_storage: ObjectStorage = None,
                 query_processor: 'QueryProcessor' = None):
        self.max_memories = max_memories
        self.memories: deque = deque(maxlen=max_memories)
        self.index: Dict[str, List[Tuple[Memory, float]]] = {}
        self.use_hybrid = use_hybrid
        self.action_storage = action_storage
        self.object_storage = object_storage
        self.query_processor = query_processor
        if self.use_hybrid:
            self.bm25_index = BM25Index()
            self.vector_index = VectorIndex()

    def add_memory(self, mem: Memory):
        if self.action_storage and self.object_storage:
            mem.ensure_extensions_applied(self.action_storage, self.object_storage)
        # Remove old index entries if memory already present (to avoid duplication)
        self._remove_from_index(mem)
        if mem not in self.memories:
            if len(self.memories) >= self.max_memories:
                oldest = self.memories[0]
                self._remove_from_index(oldest)
                if self.use_hybrid:
                    self._rebuild_hybrid_indices()
            self.memories.append(mem)
        self._index_memory(mem)
        if self.use_hybrid:
            self.bm25_index.add_memory(mem)
            self.vector_index.add_memory(mem)

    def _index_memory(self, mem: Memory):
        term_weights: Dict[str, float] = {}

        def index_action(action: Action, weight_scale: float = 1.0):
            term_weights[action.name] = term_weights.get(action.name, 0.0) + 2.0 * weight_scale
            for tag in action.tags:
                term_weights[tag] = term_weights.get(tag, 0.0) + 2.0 * weight_scale
            for obj, attr, old, new in action.physical_changes:
                term = f"{obj.name}.{attr}={new}"
                term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
            for actor, rel, target in action.physical_relations:
                term = f"{actor} {rel} {target}"
                term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
            meta = action.kinematic_metadata
            for key, val in meta.items():
                if isinstance(val, str):
                    term = f"{key}={val}"
                    term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
                elif isinstance(val, dict):
                    for k2, v2 in val.items():
                        term = f"{key}_{k2}={v2}"
                        term_weights[term] = term_weights.get(term, 0.0) + 1.5 * weight_scale
            if "bound_target" in meta and meta["bound_target"]:
                term = f"target={meta['bound_target']}"
                term_weights[term] = term_weights.get(term, 0.0) + 2.0 * weight_scale
            for sub in action.sub_actions:
                index_action(sub, weight_scale * 0.8)

        for obj in mem.objects.values():
            for attr, val in obj.attributes.items():
                if val is not None:
                    term = f"{attr}={val}"
                    term_weights[term] = term_weights.get(term, 0.0) + 1.0
                    for i, cat in enumerate(obj.categories):
                        cat_term = f"{attr}={val} ({cat})"
                        weight = 1.0 / (i + 2)
                        term_weights[cat_term] = term_weights.get(cat_term, 0.0) + weight

        for obj in mem.objects.values():
            for rel in obj.spatial_relations:
                spatial_term = f"{rel[0]} {rel[1]} {rel[2]}"
                term_weights[spatial_term] = term_weights.get(spatial_term, 0.0) + 1.0

        for act in mem.actions:
            index_action(act)

        # Index referenced sub-actions (from decomposition propagation) with lower weight
        for parent_name, sub_names in mem._referenced_sub_actions.items():
            for sub_name in sub_names:
                term = f"{sub_name} (ref)"
                term_weights[term] = term_weights.get(term, 0.0) + 1.0   # reference weight
                # Also add bare name so plain search works
                term_weights[sub_name] = term_weights.get(sub_name, 0.0) + 1.0

        for term, weight in term_weights.items():
            self.index.setdefault(term, []).append((mem, weight))

What is new / changed?
ActionStorage now has:
_usage_index (reverse mapping from template name to set of Memory objects).
register_memory_usage() / unregister_memory().
add_template() detects atomic‑to‑complex promotion and calls _propagate_new_decomposition().
Memory now has _referenced_sub_actions and handles the new "decomposition_reference" extension type without fully instantiating sub‑actions.
ActionDecomposer now contains:
deep_decompose() recursive method.
_handle_leaf_action() that tries atomic match then calls LLM, handling cyclical outputs.
_handle_complex_action() that tries exact match by flattened sub‑action list.
instantiate_reference() for lazy LLM‑based instantiation when a referenced sub‑action becomes important.
MemoryPipeline – a new class that calls deep_decompose on all top‑level actions and then registers template usage across the whole tree, ensuring the reverse index is populated.
SearchEngine indexing now also adds terms from _referenced_sub_actions (with a "(ref)" suffix and bare name) so that searches for newly learned sub‑actions can find old memories immediately.
Cyclical action support is woven into ActionTemplate/Action and the instantiation / decomposition logic.
With this code, the system will automatically enrich every existing memory when a new decomposition is discovered, and the lazy instantiation keeps it efficient. The apple‑peeling encoding can be ingested as a JSON (with the hierarchy we defined) and will be fully decomposed into atomic/cyclical actions, with all cross‑memory learning happening without user intervention.

he updated code addresses several of the earlier critical issues, but a few major pieces remain unimplemented. Here’s a breakdown of what was fixed and what is still outstanding based on your points 1–5 from the summary.

✅ Fixed Issues
1. _participant_names constructor bug
Status: Fixed.
In the original Action dataclass, _participant_names was declared with init=False, yet ActionStorage.instantiate passed it as a keyword argument, causing a TypeError.
The updated Action now uses _participant_names: List[str] = field(default_factory=list) without init=False, so it accepts the argument. This allows actions to be constructed correctly.
2. Participant resolution after decomposition
Status: Significantly improved, but with a remaining edge‑case.
The new ActionDecomposer methods (deep_decompose, _handle_leaf_action, _build_action_from_dict, _resolve_participants) now resolve participants against the memory’s object pool (object_pool parameter, which is the memory.objects dict). How to close the gap – a memory‑based planner
We can build a backward‑chaining planner that treats each complete memory as a macro‑operator. To do that we need:
Extract the net effect of a memory – the difference between the world state at the start of the memory and the world state at its end.
Extract the preconditions – all object states that must be true for the memory’s actions to succeed (we can either compute them from the actions’ preconditions, or simply record the initial state).
Index these preconditions and effects so the planner can quickly find memories whose effects match a goal.
Recursively chain: given a goal, find memories that achieve it, then treat the preconditions of those memories as new sub‑goals, and repeat.
Because the SearchEngine already indexes all attributes, we can reuse it: the planner calls search with the goal attributes as query terms, gets back ranked memories that contain those attributes in their final states, and then checks if the memory’s net effect actually matches the goal (since search is approximate). For each candidate memory, it computes the required preconditions and recurses.


Proposed implementation sketch
Step 1 – Pre‑compute memory summaries at ingestion
When a memory is ingested and fully decomposed, compute and store:
initial_state: snapshot of all object attributes before any actions.
final_state: snapshot after all actions (this is already implicitly captured by the overridden object attributes at the end).
net_effect: set of (obj_name, attr, new_value) pairs that changed from initial to final.
These can be stored directly in the Memory object.
Step 2 – Extend SearchEngine to index these summaries
In _index_memory, also index terms like effect:apple.skin_present=false with a special prefix, or index the same attribute terms we already do. The current indexing already picks up attr=value from objects, so if we just ensure that the memory’s final object states are indexed (which they are, because we index all object attributes), then a query for skin_present=false will already return peeling memories. The issue is that the planner needs to know which attributes the memory actively achieves (i.e., its effects) and which are preconditions, so we can avoid loops. We might also index precondition terms like precondition:apple.skin_present=true to allow fast look‑up of what a memory requires.
Step 3 – Build the planner
python
class MemoryBasedPlanner:
    def __init__(self, search_engine: SearchEngine, action_storage: ActionStorage, object_storage: ObjectStorage):
        self.se = search_engine
        self.action_storage = action_storage
        self.object_storage = object_storage

    def plan(self, goal_state: Dict[str, Dict[str, Any]], max_depth: int = 5) -> Optional[List[Memory]]:
        """Backward chain using memories as operators."""
        if max_depth == 0:
            return None

        # Convert goal to search terms (we can use QueryProcessor)
        goal_terms = []
        for obj_name, attrs in goal_state.items():
            for attr, val in attrs.items():
                goal_terms.append((f"{attr}={val}", 2.0))

        # Find candidate memories that mention any of these final states
        candidates = self.se.search(goal_terms)[:20]  # top 20

        for memory, score in candidates:
            # Check if the memory's actual final state satisfies the goal
            mem_final_state = self._get_final_state(memory)
            if self._state_subsumes(mem_final_state, goal_state):
                # This memory achieves the goal – now find preconditions
                mem_preconds = self._get_preconditions(memory)
                if not mem_preconds:
                    return [memory]   # no preconditions, plan complete

                # Recursively satisfy preconditions
                sub_plan = self.plan(mem_preconds, max_depth-1)
                if sub_plan is not None:
                    return sub_plan + [memory]

        return None
Key helper methods:
_get_final_state(memory) – return dict of {obj_name: {attr: value}} from the memory’s objects after all actions (i.e., current overrides).
_state_subsumes(state, goal) – check if state contains all attribute‑value pairs in goal.
_get_preconditions(memory) – computed either by replaying the actions’ preconditions or simply taking the initial object states that differ from the final (and are needed). A simple heuristic: any attribute that changed must have had a specific starting value; that starting value becomes a precondition.


Why this works with the existing index
The SearchEngine.search method uses the same inverted index that maps "skin_present=false" → list of (Memory, weight). Because we already index all object attributes (the _index_memory loop over obj.attributes.items()), any memory where the final state includes skin_present=false will be returned. So the planner can immediately leverage the indexing you already have, without any duplication.
The only new data we might want to store is the net effect list (to filter candidates faster) and precondition list (to generate sub‑goals). But even without that, the planner can work by re‑simulating the memory’s initial state from the stored object templates (since we know which attributes were overridden). It’s straightforward.

2. Building an Operator Library
We need a central registry that the planner can query to find actions (either from memories or from templates) that achieve a given set of attribute changes.
Create a new class OperatorLibrary:

class OperatorLibrary:
    def __init__(self):
        # effect signature -> list of operators that achieve it
        self.operators_by_effect: Dict[Tuple[str, str, Any], List[Operator]] = defaultdict(list)
        self.operators: List[Operator] = []   # all operators

    def add_operator(self, op: 'Operator'):
        self.operators.append(op)
        for effect in op.effects:   # each effect is (obj_name, attr, new_value)
            self.operators_by_effect[(effect[0], effect[1], effect[2])].append(op)

    def find_operators_achieving(self, obj_name: str, attr: str, value: Any) -> List['Operator']:
        return self.operators_by_effect.get((obj_name, attr, value), [])

For each action in a memory (after snapshots are captured), we can create an Operator by:
Mapping the action’s concrete participants to abstract roles. The roles can be derived from the action template’s default_participants or from the _participant_names used during instantiation. A practical middle ground: when building the operator from a memory action, we can keep the original object’s attribute requirements as preconditions. For example, the action detach_last_peel in the red‑apple memory required the apple to have color=red_brushed (because that was the apple’s color at the time). The snapshots contain the full state, but the operator’s preconditions should be the minimal set needed. We can derive this minimal set from:
The explicit action.preconditions list (which, in the current design, may be empty if not specified).
The attributes that are read but not written in the action’s changes? Not directly available.
A simpler heuristic: use the action’s known required attributes from its template (if any), plus the state of the participants at the start. But that would again be too broad.
Given the system’s desire for rich recall, we can initially store the full snapshot but filter later. For the planner, we can let the operator’s preconditions be the entire pre‑snapshot of the objects involved in the action. While this makes the operator very specific, the planner can still use it by binding planning objects that have exactly those attributes. That’s actually exactly what we want for exact memory recall: the action “peeling a red apple” only applies to red apples. However, the planner might later want to peel a green apple and should not be blocked. So we need a way to generalize operators by omitting attributes that are incidental.
Ideal solution: When an action is defined, its template specifies exactly which attributes it reads (preconditions) and writes (effects). In the peeling example, the action detach_last_peel only cares that apple.skin_present = true and apple.flesh_exposed = true (maybe). The apple’s color is irrelevant. Thus, the action’s preconditions should only list the essential attributes. In our encoding of the peeling memory, we did not explicitly define per‑action preconditions; we assumed they were inherited from the templates. So the action templates need to carry those preconditions. If they do, then when a memory instantiates the template with a red apple, the preconditions remain the generic ones (e.g., apple.skin_present=true), not color=red. The planner will then be able to apply that action to any apple with skin on, regardless of color, as long as a red apple is available.
Therefore, the correct approach is to ensure that action templates include abstract preconditions (e.g., {"object": "target_object", "attribute": "skin_present", "value": true}). The concrete memory will bind target_object to a specific apple. The Operator derived from the memory will carry those abstract preconditions, still with the bound role (e.g., role fruit requires skin_present=true). It won’t carry color=red because that was never part of the action’s formal preconditions.
Thus, we don’t need full snapshots for precondition derivation; we already have the structured preconditions from the action templates. We just need to propagate them correctly from the template to the instantiated action. Currently, ActionTemplate has a preconditions field, but Action.preconditions is of type List[Tuple[Object, str, Any]] – that’s already bound to specific objects. So for an action instantiated from a template, the preconditions are already properly scoped: they only include the attributes that the action truly requires. The fact that the apple happened to be red is not included unless the template says so. So we already have what we need.
Thus, every action in a memory already carries its own minimal preconditions and effects. The effects are the changes list. So we can build an Operator directly from each Action instance, without extra snapshots. The snapshots are only needed if we want to later infer preconditions that were not explicitly stated in the template (which is unlikely for well‑designed templates). So we can skip the snapshotting and rely on the action’s preconditions and changes.

4. The Planner that Uses the Operator Library
The planner is now a STRIPS‑like backward search:
python
class OperatorBasedPlanner:
    def __init__(self, operator_library: OperatorLibrary):
        self.library = operator_library

    def plan(self, goal: Dict[str, Dict[str, Any]], state: Dict[str, Dict[str, Any]], max_depth: int = 10) -> Optional[List[Operator]]:
        # Convert goal to list of (obj, attr, value) to achieve
        goal_attrs = set()
        for obj, attrs in goal.items():
            for attr, val in attrs.items():
                goal_attrs.add((obj, attr, val))

        # Recursive backward search
        return self._backward_search(goal_attrs, state, {}, max_depth)

    def _backward_search(self, goals, state, bindings, depth):
        if not goals:
            return []
        if depth == 0:
            return None

        # Pick one goal literal
        goal = goals.pop()
        obj_name, attr, value = goal

        # Find operators that achieve this goal
        candidates = self.library.find_operators_achieving(obj_name, attr, value)
        for op in candidates:
            # Try to bind this operator's roles to objects in the state
            new_bindings, new_state = self._apply_operator_effect(op, state, bindings, goal)
            if new_bindings is None:
                continue
            # Add operator's preconditions as new subgoals
            new_goals = goals.copy()
            for prec in op.preconditions:
                prec_obj, prec_attr, prec_val = prec
                # Apply binding to the precondition's object name
                bound_obj = new_bindings.get(prec_obj, prec_obj)
                new_goals.add((bound_obj, prec_attr, prec_val))
            # Recurse
            plan = self._backward_search(new_goals, new_state, new_bindings, depth-1)
            if plan is not None:
                return [op] + plan
        return None
The key challenge is binding: the operator’s roles (like "fruit") must be mapped to actual object names in the planning state. The planner maintains a state {obj_name: {attr: val}} and a binding map {role: obj_name}. When considering an operator, it tries to find objects in the current state that satisfy the operator’s preconditions and the specific goal literal. For instance, if the goal is to have apple_1.skin_present=false, and an operator has effect (role="target", attr="skin_present", value=false), then the planner can bind target to apple_1 if apple_1 currently exists and has skin_present=true (which would be the precondition). The planner then updates the state by applying the operator’s effects.
This means we need to define how operator roles relate to the objects. In our memory-derived operators, we can set the roles to be the original participant names (e.g., left_hand, apple). Then when we want to use this operator in a new planning context, the planner must find an existing object named apple (or create one) with the required attributes. That works if we name objects consistently. The memory’s apple is named "apple" in the object pool. So the operator’s participants are "apple", "knife", etc. The planner can treat these as concrete object identifiers, but that would limit the plan to use exactly the apple object named "apple". For multiple apples, we need variables. So we should abstract away from concrete names.
A simple method: when constructing the operator from a memory action, replace the specific object names with typed role names based on the template’s participant list. For example, if the action template peel_apple has participants [left_hand, right_hand, apple, knife, cutting_board], then the operator from a specific memory keeps those roles. The precondition apple.skin_present=true becomes (role="apple", attr="skin_present", value=true). The planner can then bind "apple" to any planning object that has the right type/attributes. This is exactly what we need.
So we will store operators with roles taken from the template’s default_participants. The planner will handle binding.

During memory ingestion, after building the action tree, traverse it and add an Operator to the OperatorLibrary for each action (including the root and all sub‑actions). The operator’s preconditions = action.preconditions (converted to role‑based form), effects = action.changes (also role‑based).
Additionally, compute a memory macro operator that summarizes the whole memory.
Object state for planning can be initialized from the current real‑world state or a hypothetical one.
The planner uses the operator library to search for a plan. It can mix operators from any memory and from action templates (templates also become operators with typed variables).
The instantiated attribute values are used naturally: if a memory action had a precondition apple.color=red (because the action template actually required a red apple), then that operator will only be applicable to red apples. If the template didn’t specify color, the operator won’t either, and any apple will do. The incidental fact that the apple in that memory happened to be red is not encoded unless the action cares about it. That’s correct.
However, in some planning contexts, the user might want to specifically use the exact apple from a memory (e.g., “use the same apple as in that peeling memory”). We can handle that by allowing two modes: generic operators (abstracted) and concrete operators (grounded to memory objects). The planner can prefer grounded ones when the object's attributes match the goal exactly, providing a direct "remembered" solution.

class Operator:

    name: str

    preconditions: List[Tuple[str, str, Any]]     # (role, attr, value)

class OperatorLibrary:
    """Stores all planning operators (from memories and templates)."""

    def __init__(self):
        # Index: (object_template_name, attr, value) -> list of operators that achieve it
        self._effect_index: Dict[Tuple[str, str, Any], List[Operator]] = defaultdict(list)
        # Index: operator name -> operator (for macro lookup)
        self._all_operators: List[Operator] = []

    def add_operator(self, op: Operator):
        self._all_operators.append(op)
        for effect in op.effects:
            role, attr, value = effect
            # Also index by the role's object type if known
            obj_type = op.role_types.get(role, role)
            self._effect_index[(obj_type, attr, value)].append(op)
            # Index by the role name itself as well (for exact matching)
            self._effect_index[(role, attr, value)].append(op)

    def find_operators_achieving(self,
                                 obj_type_or_name: str,
                                 attr: str,
                                 value: Any) -> List[Operator]:
        """Find operators whose effect includes (obj, attr, value)."""
        return self._effect_index.get((obj_type_or_name, attr, value), [])

    def find_macro_for_memory(self, memory_id: str) -> Optional[Operator]:
        """Return the macro operator representing the whole memory."""
        for op in self._all_operators:
            if op.source_memory_id == memory_id and op.is_macro:
                return op

class Memory:
    # ... existing code ...

    def build_operators(self, action_storage: ActionStorage, operator_library: OperatorLibrary):
        """Create operators for every action in this memory and add to library."""
        # 1. Build per-action operators
        def process_action(action: Action, parent_roles: List[str] = None):
            # Determine roles from the action's template
            template = action_storage.get_template(action.name)
            if template:
                roles = template.default_participants[:]
                role_types = {}
                # Map each role to the object template name of the actual participant
                for role, participant in zip(roles, action.participants):
                    role_types[role] = participant.template_name
            else:
                # fallback: use participant names as roles
                roles = [p.name for p in action.participants]
                role_types = {p.name: p.template_name for p in action.participants}

            # Build preconditions from action.preconditions (which are (Object, attr, value))
            preconditions = []
            for prec in action.preconditions:
                obj, attr, val = prec
                # Find which role corresponds to this object
                role = None
                for r, p in zip(roles, action.participants):
                    if p is obj:
                        role = r
                        break
                if role is None:
                    role = obj.name  # fallback
                preconditions.append((role, attr, val))

            # Build effects from action.changes
            effects = []
            for obj, attr, old, new in action.changes:
                role = None
                for r, p in zip(roles, action.participants):
                    if p is obj:
                        role = r
                        break
                if role is None:
                    role = obj.name
                if new is not None:
                    effects.append((role, attr, new))

            op = Operator(
                name=action.name,
                roles=roles,
                preconditions=preconditions,
                effects=effects,
                duration=int(action.effective_duration),
                source_action=action,
                source_memory_id=self.id,
                is_macro=False,
                role_types=role_types
            )
            operator_library.add_operator(op)

            # Recurse into sub-actions
            for sub in action.sub_actions:
                process_action(sub, roles)

        for action in self.actions:
            process_action(action)

        # 2. Build macro operator for the whole memory
        macro_op = self._build_macro_operator(action_storage)
        if macro_op:
            operator_library.add_operator(macro_op)

    def _build_macro_operator(self, action_storage: ActionStorage) -> Optional[Operator]:
        """Build an operator that summarizes the whole memory's effect."""
        preconds = self.get_precondition_attrs()
        effects = self.get_effect_attrs()

        if not effects:
            return None

        # Convert to role-based lists (using actual object names as roles)
        precond_list = []
        for obj_name, attrs in preconds.items():
            for attr, val in attrs.items():
                precond_list.append((obj_name, attr, val))

        effect_list = []
        for obj_name, attrs in effects.items():
            for attr, val in attrs.items():
                effect_list.append((obj_name, attr, val))

        total_duration = sum(int(a.effective_duration) for a in self.actions)

        return Operator(
            name=f"macro_{self.id}",
            roles=list(self.objects.keys()),
            preconditions=precond_list,
            effects=effect_list,
            duration=total_duration,
            source_memory_id=self.id,
            is_macro=True,
            role_types={name: obj.template_name for name, obj in self.objects.items()}
        )

3. The Operator‑Based Planner

class OperatorBasedPlanner:
    """Backward-chaining planner using operators from memories and templates."""

    def __init__(self, operator_library: OperatorLibrary, object_storage: ObjectStorage):
        self.library = operator_library
        self.object_storage = object_storage

    def plan(self,
             goal_state: Dict[str, Dict[str, Any]],
             current_state: Dict[str, Dict[str, Any]],
             max_depth: int = 10) -> Optional[List[Operator]]:
        """
        goal_state: {obj_name: {attr: value, ...}, ...}
        current_state: {obj_name: {attr: value, ...}, ...}
        Returns a sequence of operators that achieves the goal.
        """
        # Convert goal to set of literals
        goal_literals = set()
        for obj_name, attrs in goal_state.items():
            for attr, val in attrs.items():
                if current_state.get(obj_name, {}).get(attr) != val:
                    goal_literals.add((obj_name, attr, val))

        if not goal_literals:
            return []

        # Search
        plan, _ = self._backward_search(
            goal_literals=goal_literals,
            state=copy.deepcopy(current_state),
            bindings={},   # role -> obj_name
            depth=max_depth,
            visited=set()
        )
        return plan

    def _backward_search(self, goal_literals, state, bindings, depth, visited):
        if depth == 0:
            return None, None
        if not goal_literals:
            return [], state

        # Choose one goal literal (heuristic: pick the one with fewest achieving operators)
        goal = min(goal_literals, key=lambda g: len(self.library.find_operators_achieving(*g)))
        obj_name, attr, value = goal

        candidates = self.library.find_operators_achieving(obj_name, attr, value)
        # Also try object type if obj_name is a specific instance
        obj_type = self.object_storage.get_defaults(obj_name).get("__template__", obj_name)
        if obj_type != obj_name:
            candidates += self.library.find_operators_achieving(obj_type, attr, value)

        for op in candidates:
            # Try to unify operator with the current state
            new_bindings = self._unify_operator(op, state, bindings, goal)
            if new_bindings is None:
                continue

            # Apply operator effects to produce a new state (forward projection)
            new_state = self._apply_effects(op, state, new_bindings)
            if new_state is None:
                continue

            # Remove achieved literal from goals
            remaining_goals = goal_literals - {goal}
            # Also remove any other goals that are now satisfied by the new state
            remaining_goals = {g for g in remaining_goals
                               if new_state.get(g[0], {}).get(g[1]) != g[2]}

            # Add operator's preconditions as new sub-goals
            new_goals = remaining_goals.copy()
            for role, prec_attr, prec_val in op.preconditions:
                bound_obj = new_bindings.get(role, role)
                # Only add if not already true in state
                if state.get(bound_obj, {}).get(prec_attr) != prec_val:
                    new_goals.add((bound_obj, prec_attr, prec_val))

            # Create state hash for cycle detection
            state_hash = self._state_hash(new_state)
            if state_hash in visited:
                continue
            visited.add(state_hash)

            # Recursive call
            sub_plan, final_state = self._backward_search(
                new_goals, new_state, new_bindings, depth-1, visited
            )
            if sub_plan is not None:
                return [op] + sub_plan, final_state

        return None, None

    def _unify_operator(self, op: Operator, state: Dict, current_bindings: Dict, goal_literal) -> Optional[Dict]:
        """Try to find objects in state that match the operator's roles and preconditions.
           Returns extended bindings (role -> obj_name) if successful."""
        bindings = dict(current_bindings)
        obj_name, attr, value = goal_literal

        # Find which role achieves the goal
        achieving_role = None
        for role, eff_attr, eff_val in op.effects:
            if eff_attr == attr and eff_val == value:
                achieving_role = role
                break
        if achieving_role is None:
            return None

        # Bind the achieving role to the target object
        if achieving_role in bindings:
            if bindings[achieving_role] != obj_name:
                return None
        else:
            bindings[achieving_role] = obj_name

        # Verify all preconditions can be satisfied with current bindings
        for role, prec_attr, prec_val in op.preconditions:
            bound_obj = bindings.get(role)
            if bound_obj is None:
                # Find an object in state that has the required attribute value
                for s_obj, s_attrs in state.items():
                    if s_attrs.get(prec_attr) == prec_val:
                        bindings[role] = s_obj
                        bound_obj = s_obj
                        break
                if bound_obj is None:
                    # Check if we can create a new object matching the role type
                    role_type = op.role_types.get(role, role)
                    if role_type in self.object_storage.list_templates():
                        # Create a virtual object (the planner will need to instantiate it later)
                        bindings[role] = f"new_{role_type}"
                        bound_obj = bindings[role]
                    else:
                        return None
            else:
                # Verify the bound object actually satisfies the precondition
                actual_val = state.get(bound_obj, {}).get(prec_attr)
                if actual_val is not None and actual_val != prec_val:
                    return None

        return bindings

    def _apply_effects(self, op: Operator, state: Dict, bindings: Dict) -> Optional[Dict]:
        """Apply the operator's effects to a copy of the state."""
        new_state = copy.deepcopy(state)
        for role, attr, val in op.effects:
            bound_obj = bindings.get(role, role)
            if bound_obj not in new_state:
                new_state[bound_obj] = {}
            new_state[bound_obj][attr] = val
        return new_state

    def _state_hash(self, state: Dict) -> int:
        """Create a hashable representation of the state for cycle detection."""
        items = []
        for obj_name in sorted(state.keys()):
            for attr in sorted(state[obj_name].keys()):
                items.append((obj_name, attr, state[obj_name][attr]))
        return hash(tuple(items))

class HybridPlanner:
    def __init__(self, operator_library: OperatorLibrary, action_storage: ActionStorage,
                 object_storage: ObjectStorage, search_engine: SearchEngine):
        self.operator_planner = OperatorBasedPlanner(operator_library, object_storage)
        self.template_planner = TemplatePlanner(action_storage, object_storage, search_engine)
        self.search_engine = search_engine

    def plan(self, goal_state: Dict[str, Dict[str, Any]],
             initial_state: Dict[str, Dict[str, Any]],
             max_depth: int = 10) -> Optional[List[Operator]]:
        """Hybrid planning: try operator-based (memory) first, then fallback to templates."""

        # Phase 1: Use operator library (includes memory operators)
        plan = self.operator_planner.plan(goal_state, initial_state, max_depth)
        if plan is not None:
            return plan

        # Phase 2: Fallback to template-only planning
        template_plan = self.template_planner.plan_backward(goal_state, max_depth)
        if template_plan is not None:
            # Convert template actions to operators
            return self._actions_to_operators(template_plan)

        return None

    def _actions_to_operators(self, actions: List[Action]) -> List[Operator]:
        # Convert a list of actions from template planning into operators
        # for a unified plan representation.
        ops = []
        for action in actions:
            op = Operator(
                name=action.name,
                roles=[p.name for p in action.participants],
                preconditions=[(p.name, attr, val) for p, attr, val in action.preconditions],
                effects=[(p.name, attr, new) for p, attr, old, new in action.changes],
                duration=int(action.effective_duration),
                source_action=action,
                is_macro=False
            )
            ops.append(op)
        return ops

Template fallback: If no memory operator covers a sub‑goal, the template‑based planner fills the gap, and its actions are converted to operators for a unified plan.
The system now truly examines the internal structure of memories for planning, exactly as you envisioned.

class Object:
    def __init__(self, name: str, template_name: str = None, obj_id: str = None):
        self.name = name
        self.id = obj_id or f"{name}_{id(self)}"
        self.template_name = template_name or name
        self.overrides: Dict[str, Any] = {}
        self.categories: List[str] = []

    @property
    def attributes(self) -> Dict[str, Any]:
        return self.overrides

class Protagonist(Object):
    def __init__(self, name: str = "person"):
        super().__init__(name, template_name="person")
        self.overrides.update({
            "intention": None, "cheerfulness": 0.5, "focus": 0.8,
            "fatigue": 0.2, "satisfaction": 0.3,
        })

class ObjectStorage:
    _instance = None
    def __new__(cls): 
        if cls._instance is None: cls._instance = super().__new__(cls)
        return cls._instance
    def __init__(self): self._defaults = {}
    def add_template(self, name: str, defaults: Dict[str, Any]): self._defaults[name] = defaults
    def get_defaults(self, name: str) -> Dict[str, Any]: return self._defaults.get(name, {})
    def list_templates(self) -> List[str]: return list(self._defaults.keys())
    def instantiate(self, name: str, overrides: Dict[str, Any] = None) -> Object:
        # Register new objects as templates and merge new attributes
        if name not in self._defaults:
            self._defaults[name] = (overrides or {}).copy()
        else:
            if overrides:
                for k, v in overrides.items():
                    if k not in self._defaults[name]:
                        self._defaults[name][k] = v
        obj = Object(name, template_name=name)
        if overrides:
            obj.overrides = overrides.copy()
        return obj

@dataclass
class Action:
    name: str; duration: int
    participants: List[Object] = field(default_factory=list)
    changes: List[Tuple[Object, str, Any, Any]] = field(default_factory=list)
    tags: List[str] = field(default_factory=list)
    action_category: str = "INTERACTION"
    kinematic_trajectory: str = "straight"
    kinematic_metadata: Dict[str, Any] = field(default_factory=dict)
    sub_actions: List['Action'] = field(default_factory=list)
    temporal_type: str = "atomic"
    repetitions: int = 1; cycle_duration: float = 0.0
    @property
    def effective_duration(self): return self.cycle_duration * self.repetitions if self.temporal_type == "cyclical" else self.duration

class ActionStorage:
    _instance = None
    def __new__(cls): 
        if cls._instance is None: cls._instance = super().__new__(cls)
        return cls._instance
    def __init__(self): self._templates: Dict[str, dict] = {}
    def add_template(self, name: str, **kwargs): self._templates[name] = kwargs
    def get_template(self, name: str): return self._templates.get(name)
    def list_templates(self) -> List[str]: return list(self._templates.keys())
    def instantiate(self, name: str, overrides: Dict[str, Any], auto_add: bool = True) -> Action:
        if name not in self._templates:
            if auto_add:
                self._templates[name] = {"default_duration": overrides.get("duration", 1)}
            else:
                raise ValueError(f"Unknown action: {name}")
        dur = overrides.get("duration", self._templates[name].get("default_duration", 1))
        try:
            dur = int(dur)
        except (ValueError, TypeError):
            dur = self._templates[name].get("default_duration", 1)
        changes_raw = overrides.get("changes", [])
        changes = []
        for c in changes_raw:
            if isinstance(c, dict):
                changes.append((c.get("object"), c.get("attribute"), c.get("old"), c.get("new")))
            else:
                changes.append(c)
        return Action(
            name=name, duration=dur, participants=[],
            changes=changes,
            tags=overrides.get("tags", []),
            action_category=overrides.get("action_category", "INTERACTION"),
            kinematic_trajectory=overrides.get("kinematic_trajectory", "straight"),
            kinematic_metadata=overrides.get("kinematic_metadata", {}),
        )

class Memory:
    def __init__(self, memory_id: str, protagonist: Protagonist):
        self.id = memory_id; self.protagonist = protagonist
        self.objects: Dict[str, Object] = {protagonist.name: protagonist}
        self.actions: List[Action] = []
    def add_object(self, obj: Object): self.objects[obj.name] = obj
    def add_action(self, action: Action): self.actions.append(action)
    def to_dict(self) -> dict:
        objects = {}
        for name, obj in self.objects.items():
            objects[name] = {"template_name": obj.template_name, "overrides": obj.overrides}
        return {
            "memory_id": self.id,
            "protagonist": {"name": self.protagonist.name, "overrides": self.protagonist.overrides},
            "objects": objects,
            "actions": [self._action_to_dict(a) for a in self.actions]
        }
    def _action_to_dict(self, action: Action) -> dict:
        return {
            "name": action.name, "duration": action.duration,
            "participants": [p.name for p in action.participants],
            "changes": [(str(c[0]), c[1], c[2], c[3]) for c in action.changes],
            "tags": action.tags, "action_category": action.action_category,
            "kinematic_trajectory": action.kinematic_trajectory,
            "kinematic_metadata": action.kinematic_metadata,
            "sub_actions": [self._action_to_dict(s) for s in action.sub_actions],
        }

class SearchEngine:
    def __init__(self, max_memories=10000):
        self.memories = deque(maxlen=max_memories)
        self.index: Dict[str, List[Tuple[Any, float]]] = defaultdict(list)
    def add_memory(self, mem: Memory):
        if len(self.memories) >= self.memories.maxlen: self.memories.popleft()
        self.memories.append(mem)
        self._index_memory(mem)
    def _index_memory(self, mem: Memory) -> Set[str]:
        terms = {}
        for obj in mem.objects.values():
            for attr, val in obj.attributes.items():
                if val is not None: terms[f"{attr}={val}"] = 1.0
            for cat in obj.categories: terms[f"category:{cat}"] = 1.0
        for act in mem.actions:
            self._index_action(act, terms)
        for term, weight in terms.items(): self.index[term].append((mem, weight))
        return set(terms.keys())
    def _index_action(self, action: Action, terms: Dict[str, float], scale=1.0):
        terms[action.name] = terms.get(action.name, 0) + 2.0*scale
        for tag in action.tags: terms[tag] = terms.get(tag, 0) + 2.0*scale
        for sub in action.sub_actions: self._index_action(sub, terms, scale*0.8)

class PersistenceManager:
    def __init__(self, base_dir: str, milestones: Set[int] = None):
        self.base_dir = Path(base_dir)
        self.main_dir = self.base_dir / "main"
        self.snapshots_dir = self.base_dir / "snapshots"
        self.main_dir.mkdir(parents=True, exist_ok=True)
        self.snapshots_dir.mkdir(parents=True, exist_ok=True)
        self.milestones = milestones or {1, 10, 50, 100, 200, 300, 1000, 10000}
    def save_memory(self, memory: Memory, nl_text: str):
        with open(self.main_dir / "memories_with_nl.jsonl", "a", encoding="utf-8") as f:
            record = {"nl_text": nl_text, "encoding": memory.to_dict()}
            f.write(json.dumps(record, ensure_ascii=False) + "\n\n---\n")
            f.flush()
    def save_object_storage(self, obj_storage: ObjectStorage):
        with open(self.main_dir / "object_storage.json", "w", encoding="utf-8") as f:
            json.dump(obj_storage._defaults, f, indent=2, ensure_ascii=False)
            f.flush()
    def save_action_storage(self, act_storage: ActionStorage):
        with open(self.main_dir / "action_storage.json", "w", encoding="utf-8") as f:
            json.dump(act_storage._templates, f, indent=2, ensure_ascii=False)
            f.flush()
    def save_pipeline_state(self, state: dict):
        with open(self.main_dir / "pipeline_state.json", "w", encoding="utf-8") as f:
            json.dump(state, f, indent=2, ensure_ascii=False)
            f.flush()
    def save_initial_activities(self, activities: List[str]):
        with open(self.main_dir / "initial_activities.json", "w", encoding="utf-8") as f:
            json.dump(activities, f, indent=2, ensure_ascii=False)
            f.flush()
    def log_unifier_call(self, entry: dict):
        with open(self.main_dir / "unifier_log.jsonl", "a", encoding="utf-8") as f:
            f.write(json.dumps(entry, ensure_ascii=False) + "\n\n---\n")
            f.flush()
    def maybe_snapshot(self, count: int):
        if count in self.milestones:
            snap_dir = self.snapshots_dir / f"gen_{count}"
            snap_dir.mkdir(parents=True, exist_ok=True)
            for f in self.main_dir.iterdir():
                if f.is_file(): shutil.copy2(f, snap_dir / f.name)
            print(f"  📸 Snapshot at {count}")

# API Logger
# ----------------------------------------------------------------------
class APILogger:
    def __init__(self, log_path: Path):
        self.log_path = log_path
        self.log_path.parent.mkdir(parents=True, exist_ok=True)

    def log(self, purpose: str, messages: List[dict], response_content: str, usage):
        try:
            usage_dict = usage.model_dump()
        except AttributeError:
            try:
                usage_dict = dict(usage)
            except Exception:
                usage_dict = {"raw": str(usage)}
        entry = {
            "timestamp": datetime.now().isoformat(),
            "purpose": purpose,
            "messages": messages,
            "response": response_content,
            "usage": usage_dict
        }
        with open(self.log_path, "a", encoding="utf-8") as f:
            f.write(json.dumps(entry, ensure_ascii=False) + "\n")
            f.flush()

# Action Decomposer (with retry)
# ----------------------------------------------------------------------
class ActionDecomposer:
    def __init__(self, llm: AsyncOpenAI, logger: Optional[APILogger] = None):
        self.llm = llm
        self.logger = logger

    async def decompose_action(self, action: Action) -> Action:
        if action.sub_actions:
            return action
        participant_names = [p.name for p in action.participants]
        prompt = f"""Action: {action.name}
Participants: {participant_names}
Duration: {action.duration}s
Current changes: {action.changes}

Is this action primitive/atomic, or can it be broken down into more fundamental sub‑actions?
If atomic, return: {{"sub_actions": []}}
If decomposable, list the sub‑actions with fields: name, duration, participants (list of names), kinematic_trajectory, temporal_type (sequential/parallel).
Output ONLY valid JSON."""

        for attempt in range(2):
            try:
                messages = [{"role":"user","content": prompt}]
                r = await self.llm.chat.completions.create(
                    model=MODEL_NAME, messages=messages,
                    response_format={"type":"json_object"}, temperature=0.2, max_tokens=4000)
                content = r.choices[0].message.content
                usage = r.usage
                if self.logger:
                    self.logger.log("decompose", messages, content, usage)
                if not content or not content.strip():
                    continue
                data = json.loads(content)
                sub_data = data.get("sub_actions", [])
                for s in sub_data:
                    dur = s.get("duration", 1)
                    try: dur = int(dur)
                    except: dur = 1
                    sub = Action(name=s["name"], duration=dur, participants=action.participants,
                                 changes=[], tags=[s["name"]],
                                 kinematic_trajectory=s.get("kinematic_trajectory","straight"),
                                 kinematic_metadata={},
                                 temporal_type=s.get("temporal_type","sequential"))
                    sub = await self.decompose_action(sub)
                    action.sub_actions.append(sub)
                if action.sub_actions:
                    action.duration = sum(a.duration for a in action.sub_actions)
                return action  # success
            except Exception as e:
                if attempt == 1:
                    print(f"    ⚠️ Decomposer error for '{action.name}': {e}")
        return action

class MemoryGenerationPipeline:
    def __init__(self, llm: AsyncOpenAI, object_storage: ObjectStorage,
                 action_storage: ActionStorage, search_engine: SearchEngine,
                 persistence: PersistenceManager, extra_context: str = "",
                 nl_examples_text: str = "",
                 api_logger: Optional[APILogger] = None):
        self.llm = llm; self.object_storage = object_storage
        self.action_storage = action_storage; self.search_engine = search_engine
        self.persistence = persistence; self.api_logger = api_logger
        self.decomposer = ActionDecomposer(llm, api_logger)
        self.generated_count = 0; self.all_searchable_terms: Set[str] = set()
        self.new_actions: Set[str] = set(); self.new_objects: Set[str] = set()
        self.semaphore = asyncio.Semaphore(MAX_CONCURRENT)
        self.id_lock = asyncio.Lock()   # ensures unique memory IDs
        self.success_count = 0; self.fail_count = 0

        # Static NL system message (cacheable) – examples from file
        self.nl_system_msg = (
            "You are a meticulous observer of physical processes. "
            "Write a detailed procedural memory in the style of the examples below. "
            "The memory should be 300–600 words, rich in sensory details, physical attributes, "
            "object states, spatial relationships, and explicit state changes. "
            "Make it as vivid and structured as the examples.\n\n"
            + nl_examples_text
        )

        # Static JSON system message (cacheable) – long context included
        json_example = {
            "protagonist": {"name": "person", "overrides": {"intention": {"apple.skin_present": False}}},
            "objects": [
                {"name": "left_hand", "overrides": {"grip_type": "power_grasp", "motion": "rotating"}},
                {"name": "apple", "overrides": {"color": "red_brushed", "shape": "spherical", "temperature": "cool", "skin_present": True}}
            ],
            "actions": [
                {"template": "peel_apple", "duration": 60,
                 "participants": ["left_hand", "right_hand", "apple", "knife"],
                 "changes": [{"object": "apple", "attribute": "skin_present", "old": True, "new": False}],
                 "tags": ["deskinning"], "kinematic_trajectory": "spiral"}
            ]
        }
        self.json_system_msg = (
            "You are a memory‑to‑JSON encoder. Output ONLY a valid JSON object, no extra text, no markdown.\n\n"
            "Follow exactly the style of the example below – a rich, hierarchical encoding with many objects, "
            "detailed overrides, and actions that may contain sub‑actions. "
            "Include explicit state changes (attribute old→new), participants, tags, and kinematic_trajectory.\n\n"
            "Example:\n" + json.dumps(json_example, indent=2) + "\n\n"
            "Use the part below for important context. The natural language memory you will encode into JSON "
            "will be provided, marked by the title 'Memory to be encoded into Json'."
        )
        if extra_context:
            self.json_system_msg += "\n\n" + extra_context

    async def run(self, initial_prompt: str):
        print(f"\n{'='*60}\n🚀 PHASE 1: Generating {INITIAL_BATCH_SIZE} initial memories...\n{'='*60}\n")
        activities = await self._brainstorm_activities(initial_prompt, INITIAL_BATCH_SIZE)
        print(f"  📋 Generated {len(activities)} activity ideas\n")
        self.persistence.save_initial_activities(activities)
        tasks = [self._generate_one_memory(act, "initial") for act in activities]
        await asyncio.gather(*tasks, return_exceptions=True)
        print(f"\n  ✅ Phase 1 complete. Generated: {self.generated_count} memories")
        print(f"  ✅ Success: {self.success_count} | ❌ Failed: {self.fail_count}\n")
        if self.generated_count < MAX_TOTAL_MEMORIES:
            print(f"{'='*60}\n🔁 PHASE 2: Expansion (target: {MAX_TOTAL_MEMORIES})...\n{'='*60}\n")
            attempt = 0
            while self.generated_count < MAX_TOTAL_MEMORIES:
                attempt += 1
                items = self._random_pick_items()
                if not items:
                    print("  ⚠️ No items to select from. Waiting..."); await asyncio.sleep(5); continue
                print(f"  🎲 Attempt {attempt}: Selected {len(items)} terms: {', '.join(items[:3])}...")
                proposed = await self._discover_unifying_activities(items)
                log_entry = {"timestamp": datetime.now().isoformat(), "selected_terms": items,
                             "proposed_activities": proposed, "generated_memories": []}
                if proposed:
                    # Limit to at most 3 activities per random selection
                    selected_activities = proposed[:3]
                    print(f"    💡 Found {len(proposed)} activities, using top {len(selected_activities)}")
                    tasks = [self._generate_one_memory(act["activity"], "composite", log_entry) for act in selected_activities]
                    await asyncio.gather(*tasks, return_exceptions=True)
                else:
                    print("    ❌ No valid unifying activities found")
                self.persistence.log_unifier_call(log_entry)
                if attempt % 10 == 0:
                    print(f"\n  📊 Progress: {self.generated_count}/{MAX_TOTAL_MEMORIES}\n")
        print(f"\n🏁 Pipeline finished! Total: {self.generated_count}, Success: {self.success_count}, Failed: {self.fail_count}")

    async def _brainstorm_activities(self, prompt: str, count: int) -> List[str]:
        context = f"Recent actions: {', '.join(list(self.new_actions)[-10:])}\nRecent objects: {', '.join(list(self.new_objects)[-10:])}\n"
        try:
            messages = [
                {"role":"system","content": 'Output ONLY valid JSON: {"activities": ["activity1", "activity2", ...]}. Each activity must be a simple string (e.g. "Wash dishes").'},
                {"role":"user","content": f"{prompt}\n{context}\nGenerate {min(count,50)} activities."}
            ]
            r = await self.llm.chat.completions.create(
                model=MODEL_NAME, messages=messages,
                response_format={"type":"json_object"}, temperature=0.8, max_tokens=20000)
            content = r.choices[0].message.content
            usage = r.usage
            if self.api_logger:
                self.api_logger.log("brainstorm", messages, content, usage)
            data = json.loads(content) if isinstance(content, str) else content
            if isinstance(data, dict) and "activities" in data:
                return [a if isinstance(a, str) else a.get("name", str(a)) for a in data["activities"]]
            return []
        except Exception as e:
            print(f"  ❌ Brainstorm failed: {e}"); return []

    async def _generate_one_memory(self, activity: str, domain: str, log_entry: dict = None):
        async with self.semaphore:
            if self.generated_count >= MAX_TOTAL_MEMORIES: return

            nl = None
            for retry in range(3):
                try:
                    nl = await self._write_nl_memory(activity)
                    if nl and len(nl) > 50: break
                    await asyncio.sleep(2 * (retry+1))
                except Exception as e:
                    print(f"    ❌ NL error (retry {retry+1}): {e}")
                    await asyncio.sleep(2 * (retry+1))
            if not nl:
                print(f"    ❌ [{activity}] NL generation failed"); self.fail_count += 1; return
            print(f"    ✓ [{activity}] NL ({len(nl)} chars)")

            json_data = await self._nl_to_json(nl)
            if not json_data:
                print(f"    ❌ [{activity}] JSON encoding failed"); self.fail_count += 1; return
            print(f"    ✓ [{activity}] JSON (objects:{len(json_data.get('objects',[]))}, actions:{len(json_data.get('actions',[]))})")

            # Assign unique memory ID under lock
            async with self.id_lock:
                memory_id = f"mem_{self.generated_count:06d}"
                self.generated_count += 1
            json_data["memory_id"] = memory_id

            try:
                memory = await self._ingest(json_data)
                self.success_count += 1
                self._post_process(memory, nl)
                print(f"    ✅ [{activity}] Memory created ({memory.id})")
                if log_entry is not None:
                    log_entry["generated_memories"].append({"activity": activity, "memory_id": memory.id, "success": True})
            except Exception as e:
                self.fail_count += 1
                print(f"    ❌ [{activity}] EXCEPTION: {type(e).__name__}: {e}")

    async def _write_nl_memory(self, activity: str) -> Optional[str]:
        obj_list = ', '.join(list(self.new_objects)[-10:]) if self.new_objects else "none"
        act_list = ', '.join(list(self.new_actions)[-10:]) if self.new_actions else "none"
        user_msg = f"Write a memory of: {activity}\n\nAvailable objects: {obj_list}\nAvailable actions: {act_list}\n"
        messages = [
            {"role":"system","content": self.nl_system_msg},
            {"role":"user","content": user_msg}
        ]
        try:
            r = await self.llm.chat.completions.create(
                model=MODEL_NAME, messages=messages,
                temperature=0.7, max_tokens=8000)
            content = r.choices[0].message.content
            if self.api_logger:
                self.api_logger.log("nl_generation", messages, content, r.usage)
            return content if isinstance(content, str) and len(content) > 50 else None
        except Exception as e:
            return None

    async def _nl_to_json(self, nl_text: str) -> Optional[dict]:
        user_msg = f"Memory to be encoded into Json:\n\n{nl_text}"
        messages = [
            {"role":"system","content": self.json_system_msg},
            {"role":"user","content": user_msg}
        ]
        for attempt in range(3):
            try:
                r = await self.llm.chat.completions.create(
                    model=MODEL_NAME, messages=messages,
                    response_format={"type":"json_object"} if attempt == 0 else None,
                    temperature=0.1, max_tokens=120000)
                content = r.choices[0].message.content.strip()
                if self.api_logger:
                    self.api_logger.log("json_encoding", messages, content, r.usage)
                if content.startswith("```json"): content = content[7:]
                if content.endswith("```"): content = content[:-3]
                if not content: continue
                data = json.loads(content)
                if not isinstance(data, dict): continue
                if "protagonist" not in data: data["protagonist"] = {"name":"person","overrides":{}}
                if "objects" not in data: data["objects"] = []
                if "actions" not in data: data["actions"] = []
                data["objects"] = [o for o in data["objects"] if isinstance(o, dict) and "name" in o]
                for o in data["objects"]:
                    if "overrides" not in o or not isinstance(o["overrides"], dict): o["overrides"] = {}
                valid_actions = []
                for a in data["actions"]:
                    if isinstance(a, dict) and "template" in a:
                        if "participants" not in a: a["participants"] = []
                        if "changes" not in a: a["changes"] = []
                        if "duration" not in a: a["duration"] = 3
                        if "tags" not in a: a["tags"] = []
                        if "kinematic_trajectory" not in a: a["kinematic_trajectory"] = "straight"
                        if "sub_actions" in a: a["sub_actions"] = self._validate_sub_actions(a["sub_actions"])
                        valid_actions.append(a)
                data["actions"] = valid_actions
                return data
            except Exception as e:
                print(f"    ⚠️ JSON attempt {attempt+1}: {e} | First 150 chars: {content[:150] if 'content' in locals() else 'N/A'}")
        return None

    def _validate_sub_actions(self, sub_actions: list) -> list:
        valid = []
        for sa in sub_actions:
            if not isinstance(sa, dict) or "template" not in sa: continue
            if "participants" not in sa: sa["participants"] = []
            if "changes" not in sa: sa["changes"] = []
            if "duration" not in sa: sa["duration"] = 1
            if "tags" not in sa: sa["tags"] = []
            if "kinematic_trajectory" not in sa: sa["kinematic_trajectory"] = "straight"
            if "sub_actions" in sa: sa["sub_actions"] = self._validate_sub_actions(sa["sub_actions"])
            valid.append(sa)
        return valid

    async def _ingest(self, json_data: dict) -> Memory:
        protag = json_data.get("protagonist", {})
        name = protag.get("name", "person") if isinstance(protag, dict) else "person"
        protagonist = Protagonist(name)
        mem = Memory(json_data.get("memory_id", f"mem_{self.generated_count:06d}"), protagonist)
        for od in json_data.get("objects", []):
            obj = self.object_storage.instantiate(od.get("name","object"), od.get("overrides",{}))
            mem.add_object(obj)
        for ad in json_data.get("actions", []):
            action = self.action_storage.instantiate(ad["template"], ad, auto_add=True)
            participants = []
            for pname in ad.get("participants", []):
                if pname in mem.objects: participants.append(mem.objects[pname])
                else:
                    obj = Object(pname); mem.objects[pname] = obj; participants.append(obj)
            action.participants = participants
            if "sub_actions" in ad: action.sub_actions = self._build_sub_actions(ad["sub_actions"], mem)
            mem.add_action(action)
        for i, action in enumerate(mem.actions):
            mem.actions[i] = await self.decomposer.decompose_action(action)
        self.search_engine.add_memory(mem)
        return mem

    def _build_sub_actions(self, sub_data: list, mem: Memory) -> List[Action]:
        sub_actions = []
        for sa in sub_data:
            action = self.action_storage.instantiate(sa["template"], sa, auto_add=True)
            participants = []
            for pname in sa.get("participants", []):
                if pname in mem.objects: participants.append(mem.objects[pname])
                else:
                    obj = Object(pname); mem.objects[pname] = obj; participants.append(obj)
            action.participants = participants
            if "sub_actions" in sa: action.sub_actions = self._build_sub_actions(sa["sub_actions"], mem)
            sub_actions.append(action)
        return sub_actions

    def _post_process(self, memory: Memory, nl_text: str):
        terms = self.search_engine._index_memory(memory)
        self.all_searchable_terms.update(terms)
        for a in memory.actions:
            self.new_actions.add(a.name)
            for s in a.sub_actions: self.new_actions.add(s.name)
        for obj in memory.objects.values(): self.new_objects.add(obj.template_name)
        self.persistence.save_memory(memory, nl_text)
        self.persistence.save_object_storage(self.object_storage)
        self.persistence.save_action_storage(self.action_storage)
        self.persistence.save_pipeline_state(self._pipeline_state())
        self.persistence.maybe_snapshot(self.generated_count)

    def _pipeline_state(self) -> dict:
        return {"generated_count": self.generated_count, "success_count": self.success_count,
                "fail_count": self.fail_count, "all_searchable_terms": list(self.all_searchable_terms),
                "new_actions": list(self.new_actions), "new_objects": list(self.new_objects)}

    def _random_pick_items(self) -> List[str]:
        if len(self.all_searchable_terms) < 2: return []
        k = min(random.randint(2, 10), len(self.all_searchable_terms))
        return random.sample(list(self.all_searchable_terms), k)

    async def _discover_unifying_activities(self, items: List[str]) -> List[dict]:
        prompt = f"""Given these concepts: {', '.join(items)}

Propose up to 10 distinct, goal‑directed household or garden activities that naturally involve ALL of these concepts.
Describe each activity in plain English, like "Peeling a red apple with a sharp knife over a cutting board".
Do not use raw search terms; write a natural, coherent task description.
Each activity must represent a single coherent goal directed activity using the concepts given. It must not be a bunch of unrelated activities grouped together into one event. For example, given the concepts Tool_usage, color=cyan, knee_flex, a single goal directed activity could be "Physical therapy session using a cyan-coloured resistance band for assisted knee flexion". It should not be unrelated activities each calling on one concept and then grouped together into one "activity". Such an amalgamation should receive a low plausibility score, for example: using a hammer to put in some nails on a shed in the backyard, while a neighbourhood child was playing with a cyan colored toy, afterwards taking a ballet dance class which involved knee flex. This disjointed activity proposal would receive a much lower score than the previous one with the resistance band exercise.

The score should be about how plausible the proposed activity is and also how coherent it is as a single activity rather than a collection of unrelated activities.

Output ONLY valid JSON: {{"activities": [{{"activity": "...", "plausibility": 0.8}}]}} (plausibility 0-1, >=0.6)."""
        for attempt in range(2):
            try:
                messages = [{"role":"user","content": prompt}]
                r = await self.llm.chat.completions.create(
                    model=MODEL_NAME, messages=messages,
                    response_format={"type":"json_object"} if attempt == 0 else None,
                    temperature=0.8, max_tokens=10000)
                content = r.choices[0].message.content.strip()
                if self.api_logger:
                    self.api_logger.log("unifier", messages, content, r.usage)
                if not content: continue
                data = json.loads(content)
                if isinstance(data, dict) and "activities" in data:
                    return [a for a in data["activities"] if isinstance(a, dict) and a.get("plausibility",0) >= 0.6][:10]
            except Exception:
                continue
        print("    ⚠️ Unifier failed after retries")
        return []
